  43def _pack_results(results: list, names: list[str] | None | bool):
  44    """Pack worker results into dict, list, or single value.
  45
  46    ``names=True`` forces list return (backward compat for old list API).
  47    ``names=None`` unwraps to single if ``len(results) == 1``.
  48    ``names=list[str]`` returns ``dict`` keyed by name.
  49    """
  50    if names is True:
  51        return results
  52    if names is not None:
  53        return dict(zip(names, results))
  54    if len(results) == 1:
  55        return results[0]
  56    return results

  89    Bundles one model level: (Model, ds, ModelRep).

  96    different nodes in a MetaModel use disjoint scope sets.
  97    """

 107    def __init__(
 108        self,
 109        model=None,
 110        ds: pd.DataFrame = None,
 111        model_path: str | None = None,
 112        options_path: str | None = None,
 113        model_rep: ModelRep | None = None,
 114        dt_pred: pd.DataFrame | None = None,
 115        missed_stop: bool = False,
 116        scope_names: set[str] | None = None,
 117        static: bool = False,
 118        _display_only: bool = False,
 119        **model_rep_kwargs,
 120    ):
 121        """
 122        Initialise a ModelNode.
 123
 124        Bundles a trained model, its dataset, and reporting settings into
 125        a single computation unit. ModelRep may be passed explicitly or
 126        auto-constructed from keyword arguments (``language``,
 127        ``trans_dict``, etc.).
 128
 129        Parameters
 130        ----------
 131        model : object, optional
 132            Trained fermatrica model instance. Optional when ``model_path``
 133            is provided. Required when ``model_path`` is ``None``.
 134        ds : pd.DataFrame, optional
 135            Dataset used for transform and predict. Optional when
 136            ``model_path`` is provided and the model directory contains a
 137            saved dataset. When ``model_path`` is given, an explicit
 138            ``ds`` **overrides** the saved (post-transform) dataset — pass
 139            the raw data so the no-arg refresh (``node.compute()``)
 140            transforms from raw.
 141        model_path : str or None, optional
 142            Path to a saved model directory. When provided, the model and
 143            dataset are auto-loaded via ``model_load_ext``. Cannot be used
 144            together with ``model``.
 145        options_path : str or None, optional
 146            Path to an options XLSX file. When provided and
 147            ``model_rep.media_prices`` is unset, the optional ``media_prices``
 148            sheet is loaded automatically. An already supplied
 149            ``model_rep.media_prices`` value is preserved.
 150        missed_stop : bool, default False
 151            Propagated to ``model_load_ext`` when ``model_path`` is used.
 152        scope_names : set[str] or None, optional
 153            Names of scopes this node participates in. When ``None``
 154            (default), all scopes passed to ``translate()`` /
 155            ``compute_after_translate()`` apply. When set, only scopes whose
 156            ``scope_name`` is in this set are translated and summarized —
 157            others are skipped. Useful for multi-node MetaModels where
 158            different nodes use disjoint scope sets (e.g. per-target UBA vs
 159            per-region sales scopes).
 160        static : bool, default False
 161            Node whose model has no option-driven media RHS. Its
 162            ``translate()`` is a no-op and no ``trans_dict`` is required,
 163            but the node still computes per option — it may consume dynamic
 164            upstream predictions via DAG merges.
 165        _display_only : bool, default False
 166            Internal bypass for visual-only nodes that carry a ``ModelRep``
 167            but no model (constructed by ``_convert_to_metamodel`` for
 168            plot dispatch). Such nodes must never be computed.
 169        model_rep : ModelRep or None, optional
 170            Pre-built reporting configuration. If ``None``, one is created
 171            from ``**model_rep_kwargs`` using ``ModelRep(ds, **kwargs)``.
 172        dt_pred : pd.DataFrame or None, optional
 173            Pre-computed prediction DataFrame. If ``None``, prediction
 174            will be computed on first call to ``compute()``.
 175        **model_rep_kwargs
 176            Keyword arguments forwarded to ``ModelRep`` when
 177            ``model_rep`` is ``None``, e.g. ``language``, ``trans_dict``.
 178
 179        Raises
 180        ------
 181        FermatricaError
 182            If both ``model_path`` and ``model`` are provided, if ``ds``
 183            cannot be resolved, or if neither a model nor ``model_path``
 184            is given (unless ``_display_only`` is set).
 185        """
 186        if model_path is not None and model is not None:
 187            fermatrica_error("Cannot specify both model_path and model")
 188
 189        if model is None and model_path is None and not _display_only:
 190            fermatrica_error(
 191                "ModelNode requires a model — pass model= or model_path="
 192            )
 193
 194        if model_path is not None:
 195            model, loaded_ds, _ = model_load_ext(model_path, missed_stop=missed_stop)
 196            if ds is None:
 197                ds = loaded_ds
 198
 199        if ds is None:
 200            fermatrica_error("ds must be provided")
 201
 202        self.model = model
 203        self.ds = ds.copy()
 204        self.dt_pred = dt_pred
 205        self.scope_names = set(scope_names) if scope_names is not None else None
 206        self.static = static
 207        self._display_only = _display_only
 208        if model_rep is not None:
 209            self.model_rep = model_rep
 210        else:
 211            self.model_rep = ModelRep(ds, **model_rep_kwargs)
 212        if options_path is not None and getattr(self.model_rep, "media_prices", None) is None:
 213            from fermatrica_rep.options.define import media_prices_read
 214
 215            self.model_rep.media_prices = media_prices_read(options_path)
 216        self._init_finish()

 242    @classmethod
 243    def load(cls, path: str, model_rep_path: str | None = None):
 244        """
 245        Load a ModelNode from a saved model directory on disk.
 246
 247        Reads the model and dataset via ``model_load_ext`` and the
 248        ModelRep from ``model_rep.pkl.lzma`` in the same directory
 249        (or a custom path). ``model_rep.pkl.lzma`` is optional — when
 250        absent (modelling-stage saves) a default ModelRep is created
 251        from the dataset (English, palettes derived from ``ds``) and a
 252        warning is logged.
 253
 254        Parameters
 255        ----------
 256        path : str
 257            Directory path containing the saved model artifacts
 258            (as produced by :meth:`ModelNode.save`).
 259        model_rep_path : str or None, optional
 260            Explicit path to the ``model_rep.pkl.lzma`` file. When
 261            ``None`` (default), looks for ``model_rep.pkl.lzma``
 262            inside ``path``.
 263
 264        Returns
 265        -------
 266        ModelNode
 267            Fully initialised node with model, dataset, and ModelRep.
 268        """
 269        model, ds, _ = model_load_ext(path, missed_stop=False)
 270
 271        if model_rep_path is None:
 272            model_rep_path = os.path.join(path, "model_rep.pkl.lzma")
 273        if os.path.exists(model_rep_path):
 274            model_rep = read_pickle_compat(model_rep_path)
 275        else:
 276            from fermatrica_rep.meta_model.meta_model import _auto_model_rep
 277
 278            logging.warning(
 279                f"ModelRep not found at {model_rep_path} — using default ModelRep"
 280            )
 281            model_rep = _auto_model_rep(ds, None)
 282
 283        node = cls(model=model, ds=ds, model_rep=model_rep)
 284
 285        toml_path = os.path.join(path, "meta_conf.toml")
 286        if os.path.exists(toml_path):
 287            with open(toml_path, "rb") as handle:
 288                conf = tomllib.load(handle)
 289            settings = conf.get("node_settings", {})
 290        else:
 291            settings = {}
 292
 293        if len(settings) == 1:
 294            entry = next(iter(settings.values()))
 295            if entry is not None and "scope_names" in entry:
 296                node.scope_names = set(entry["scope_names"])
 297            if entry is not None and "static" in entry:
 298                node.static = bool(entry["static"])
 299
 300        return node

 322    def compute(
 323        self,
 324        option=None,
 325        scopes: list[OptionScope] | None = None,
 326        allow_past: bool = False,
 327        if_exact: bool = False,
 328        *,
 329        return_full: bool = False,
 330    ) -> pd.DataFrame | dict[str, dict | tuple[pd.DataFrame, pd.DataFrame, dict]]:
 331        """
 332        Refresh the model fit, or translate budgets and derive summaries.
 333
 334        When ``option`` is ``None`` (default), runs a single
 335        transform + predict on ``self.ds`` with ``set_start=False``
 336        (refresh — the initial ``transform(set_start=True)`` bootstrap
 337        remains a core modelling step) and stores the results on the node.
 338        ``scopes`` / ``return_full`` are ignored on this path.
 339
 340        With an ``option``, translates budgets into media variables for all
 341        scopes, then runs a single transform + predict and derives per-scope
 342        summaries. Convenience method that chains :meth:`translate` and
 343        :meth:`compute_after_translate`. When used directly on a single
 344        ``ModelNode``, scopes are translated cumulatively — each scope inherits
 345        the previous scope's media values and adstock. This is the standard
 346        entry point; call the two sub-methods separately only when the caller
 347        needs to interleave custom operations between translation and compute.
 348
 349        ``self.ds`` and ``self.dt_pred`` are always updated after the
 350        transform + predict, regardless of ``return_full``.
 351
 352        Args:
 353            option: Option object with budget allocations keyed by scope name.
 354                Must provide ``iter_scopes()`` yielding ``(scope_name,
 355                budget_items)`` tuples. ``None`` runs the no-arg refresh path.
 356            scopes: List of ``OptionScope`` objects defining date ranges,
 357                entity filters, and other scope parameters. Scope names are
 358                matched against ``option.iter_scopes()`` during translation.
 359            allow_past: If ``True``, the scope mask built by
 360                ``OptionScope.build_mask`` includes historical periods
 361                (``listed`` ≤ 3). Default ``False`` restricts to ``listed ==
 362                4`` only.
 363            if_exact: If ``True``, skip long-term translation beyond the scope
 364                end date. Use for exact-period scenarios where post-scope media
 365                should not be projected forward.
 366            return_full: If ``True``, each scope entry is a 3-tuple
 367                ``(ds, dt_pred, summary)``. If ``False`` (default), each
 368                scope entry is just the ``summary`` dict. Follows the same
 369                convention as ``fermatrica.fit_predict``.
 370
 371        Returns:
 372            When ``option`` is ``None``: the ``dt_pred`` DataFrame produced by
 373            the refresh (also stored on ``self.dt_pred``).
 374
 375            With an ``option``: dict mapping ``{scope_name: summary}`` when
 376            ``return_full=False`` (default), or
 377            ``{scope_name: (ds, dt_pred, summary)}`` when ``return_full=True``.
 378
 379            ``ds`` is the transformed dataset (a copy made after
 380            transform/predict — safe to inspect without mutating
 381            ``self.ds``). ``dt_pred`` is the full prediction DataFrame
 382            (shared across all scopes — only the per-scope summary slices
 383            differ). ``summary`` is a dict with keys such as
 384            ``pred_exact_vol``, ``pred_long_vol``, ``growth_vol``, etc.
 385        """
 386        if option is None:
 387            return self._refresh()
 388        self.translate(option, scopes, allow_past=allow_past, if_exact=if_exact)
 389        return self.compute_after_translate(option, scopes, return_full=return_full)

 391    def _refresh(self) -> pd.DataFrame:
 392        """
 393        Run transform (``set_start=False``) + predict and store the result.
 394
 395        Uses the node's current ``ds`` as the transform input, then stores
 396        the transformed dataset and prediction back on the node. Returns the
 397        ``dt_pred`` DataFrame.
 398
 399        ``_check_price_var`` / ``_load_adhoc_code`` are idempotent — safe to
 400        re-run even when already executed during :meth:`translate`.
 401        """
 402        if not getattr(self.model.obj, 'models', None) or 'main' not in self.model.obj.models:
 403            fermatrica_error(
 404                "Model is not fitted — run `fm.fit_predict(ds, model, if_fit=True)` "
 405                "before `node.compute()`"
 406            )
 407        self._check_price_var(self.model.conf, self.ds)
 408        self._load_adhoc_code()
 409        model, ds = transform(ds=self.ds, model=self.model, set_start=False, if_by_ref=False)
 410        dt_pred = predict_ext(model, ds)
 411        self.ds = ds
 412        self.dt_pred = dt_pred
 413        return dt_pred

 415    def translate(
 416        self,
 417        option,
 418        scopes: list[OptionScope] | None = None,
 419        allow_past: bool = False,
 420        if_exact: bool = False,
 421    ):
 422        """
 423        Apply per-scope budget translations and media seasonality into
 424        ``self.ds``.
 425
 426        Iterates scopes sorted by ``OptionScope.date_start`` (not by the
 427        order returned by ``option.iter_scopes()``), applying each scope's
 428        seasonality correction and translating budget values into media
 429        variables directly on ``self.ds``. No copies are made between scopes
 430        — later scopes **cumulatively inherit** the translated media values
 431        and adstock carryover from earlier scopes.
 432
 433        Sorting by start date matters for ``_translate_long``: each scope's
 434        budget is projected forward beyond its ``date_end``, so a scope with
 435        an earlier period would otherwise overwrite the media of a later
 436        scope when its ``_translate_long`` runs after it. Processing scopes
 437        chronologically ensures the latest (target) scope's translation and
 438        long-term projection win.
 439
 440        No transform or predict is performed (fast, mutations only). Designed
 441        as Phase 1 of :meth:`MetaModel.compute` so predecessor ``dt_pred``
 442        merges happen before any heavy computation.
 443
 444        Raises ``FermatricaError`` at computation time when the option
 445        carries budgets for the node's scopes but ``model_rep.trans_dict``
 446        is not set — budgets would otherwise silently remain untranslated.
 447        Empty budget items are allowed without a ``trans_dict``.
 448
 449        Args:
 450            option: Option object with budget allocations keyed by scope name.
 451            scopes: List of ``OptionScope`` objects. Each scope is looked up
 452                by name; unmatched scope entries in the option are silently
 453                skipped.
 454            allow_past: Forwarded to ``OptionScope.build_mask``. When
 455                ``False`` (default), only future rows (``listed == 4``)
 456                receive translated values.
 457            if_exact: If ``True``, skip :meth:`_translate_long` — budget
 458                values are applied only to the exact scope period, not
 459                projected forward.
 460
 461        For a static node this is a no-op: its model has no media RHS, so
 462        budgets and seasonality do not apply to it.
 463        """
 464        if self.static:
 465            return
 466        model_conf = self.model.conf
 467        self._check_price_var(model_conf, self.ds)
 468        self._load_adhoc_code()
 469
 470        bdg_ssn_cols = self._build_bdg_ssn_cols()
 471
 472        # Sort scopes chronologically by their start date so that a scope
 473        # translated later (with a later date_start) is not overwritten by
 474        # _translate_long of an earlier-ending scope processed after it.
 475        scopes_sorted: list[tuple[OptionScope, dict[str, float]]] = []
 476
 477        for scope_name, budget_items in option.iter_scopes():
 478            if self.scope_names is not None and scope_name not in self.scope_names:
 479                continue
 480            scope = self._find_scope(scopes, scope_name)
 481            if scope is None:
 482                logging.warning(
 483                    f"Scope '{scope_name}' not found in provided scopes — skipping"
 484                )
 485                continue
 486            scopes_sorted.append((scope, budget_items))
 487
 488        scopes_sorted.sort(key=lambda x: x[0].date_start)
 489
 490        if self.model_rep.trans_dict is None and any(
 491            budget_items for _, budget_items in scopes_sorted
 492        ):
 493            missing_scopes = [
 494                s.scope_name for s, budget_items in scopes_sorted if budget_items
 495            ]
 496            fermatrica_error(
 497                "No translation dictionary (trans_dict) is set on the node, but "
 498                f"budgets are provided for scope(s): {', '.join(missing_scopes)}. "
 499                "Budgets cannot be translated into media variables. "
 500                "Wire a translation sheet via MetaModel(compute_context=...) "
 501                "or set node.model_rep.trans_dict before compute()."
 502            )
 503
 504        for scope, budget_items in scopes_sorted:
 505            self.ds = self._apply_media_ssn(self.ds, scope, bdg_ssn_cols)
 506            ds_mask = scope.build_mask(self.ds, allow_past=allow_past)
 507            self.ds = self._translate_budget(self.ds, model_conf, budget_items, ds_mask, scope)
 508
 509            if not if_exact:
 510                self.ds = self._translate_long(self.ds, model_conf, budget_items, scope, allow_past)
 511
 512    def compute_after_translate(
 513        self,
 514        option,
 515        scopes: list[OptionScope] | None = None,
 516        *,
 517        return_full: bool = False,
 518    ) -> dict[str, dict | tuple[pd.DataFrame, pd.DataFrame, dict]]:
 519        """
 520        Run a single transform + predict on ``self.ds``, then derive
 521        per-scope summaries from the shared prediction.
 522
 523        Assumes ``self.ds`` has already been mutated by :meth:`translate`
 524        (or equivalent cumulative translation). Runs ``transform`` and
 525        ``predict_ext`` exactly once, then slices the single ``dt_pred``
 526        via each scope's ``date_start`` / ``date_end`` and entity filters
 527        to produce per-scope summaries.
 528
 529        Designed as Phase 2 of :meth:`MetaModel.compute` — called after
 530        all predecessor ``dt_pred`` values have been merged via edge-specific
 531        merge functions.
 532
 533        ``self.ds`` and ``self.dt_pred`` are always updated, regardless of
 534        ``return_full``.
 535
 536        Args:
 537            option: Option object. Only used for ``iter_scopes()`` to
 538                enumerate scope names and budget items for summarization.
 539            scopes: List of ``OptionScope`` objects defining date ranges,
 540                entity filters, and reference periods for summaries.
 541            return_full: If ``True``, each scope entry is a 3-tuple
 542                ``(ds, dt_pred, summary)``. If ``False`` (default), each
 543                scope entry is just the ``summary`` dict.
 544
 545        Returns:
 546            dict mapping ``{scope_name: summary}`` when ``return_full=False``
 547            (default), or ``{scope_name: (ds, dt_pred, summary)}`` when
 548            ``return_full=True``.
 549
 550            ``ds`` is a **copy** of the transform output (safe to inspect).
 551            ``dt_pred`` is the full prediction DataFrame (identical for
 552            every scope — shared reference). ``summary`` is a dict with
 553            keys ``pred_exact_vol``, ``pred_long_vol``,
 554            ``pred_ref_vol``, ``growth_vol``, and their ``_val``
 555            counterparts when a price variable is defined.
 556        """
 557        dt_pred = self._refresh()
 558        ds = self.ds
 559
 560        results = {}
 561        model_conf = self.model.conf
 562        for scope_name, budget_items in option.iter_scopes():
 563            if self.scope_names is not None and scope_name not in self.scope_names:
 564                continue
 565            scope = self._find_scope(scopes, scope_name)
 566            if scope:
 567                summary = self._summarize(model_conf, budget_items, scope, dt_pred)
 568                if return_full:
 569                    results[scope_name] = (ds.copy(), dt_pred, summary)
 570                else:
 571                    results[scope_name] = summary
 572
 573        return results

 575    def _find_scope(
 576        self,
 577        scopes: list[OptionScope] | None,
 578        scope_name: str,

 695    def _load_adhoc_code(self):
 696        """
 697        Import adhoc user modules registered on
 698        ``model_rep.adhoc_code_src``.
 699
 700        Adhoc modules may define custom translator functions
 701        (``adhoc_*``) or merge functions that are referenced by name in
 702        the translation dictionary or model graph. These must be imported
 703        into the current module's namespace before dispatch.
 704
 705        Idempotency is keyed on the process-local module globals, not on
 706        per-node instance state: ``import_module_from_string`` registers the
 707        full dotted module name as a key in this module's ``globals()``, so
 708        ``k in globals()`` means "already imported in this process". Gating on
 709        the globals keeps the check correct when a node crosses a process
 710        boundary — a spawn Pool worker unpickles the node but starts with a
 711        fresh node.py namespace, so adhoc modules must be re-imported there.
 712
 713        The imported source is this node's pickled snapshot (``adhoc_code_src``
 714        captured at ``ModelRep`` construction), not the live project adhoc
 715        files. Nodes of one MetaModel may have been saved at different
 716        development stages, so the same adhoc module can be imported from a
 717        different (older) source than the current ``code_py/adhoc/`` code —
 718        check that saved sources are up to date in all nodes before relying on
 719        a loaded model.
 720
 721        Does nothing if ``adhoc_code_src`` is empty or unset.
 722        """
 723        if (
 724            pd.notna(self.model_rep.adhoc_code_src)
 725            and len(self.model_rep.adhoc_code_src) > 0
 726        ):
 727            for k, v in self.model_rep.adhoc_code_src.items():
 728                if k in globals():
 729                    continue
 730                fr_cur_name = inspect.currentframe().f_globals["__name__"]
 731                import_module_from_string(k, v, fr_cur_name)

 733    def _translate_budget(
 734        self,
 735        ds: pd.DataFrame,
 736        model_conf: ModelConf,
 737        budget_items: dict[str, float],
 738        ds_mask: pd.Series,
 739        scope: OptionScope | None = None,
 740    ) -> pd.DataFrame:
 741        """
 742        Dispatch budget items to their registered translator functions.
 743
 744        For each budget item (e.g. ``"bdg_tv": 50``):
 745
 746        1. Looks up the translator function name and target variable
 747           from ``model_rep.trans_dict``.
 748        2. Resolves the effective media price by combining the base
 749           price from ``trans_dict`` with per-scope or per-tool
 750           inflation multipliers.
 751        3. Calls the translator function via ``eval()``, passing the
 752           computed ``media_price`` and the scope mask.
 753
 754        Budget items not registered in ``trans_dict["function"]`` are skipped.
 755        Items whose translator functions cannot be resolved are skipped with a
 756        one-time warning (per unique function name, per process).
 757
 758        Parameters
 759        ----------
 760        ds : pd.DataFrame
 761            Dataset to modify in place. Media columns receive budget
 762            values on rows where ``ds_mask`` is ``True``.
 763        model_conf : ModelConf
 764            Model configuration (passed through to translator functions).
 765        budget_items : dict[str, float]
 766            Budget items keyed by item name, e.g.
 767            ``{"bdg_tv": 50.0}``. Values are in millions.
 768        ds_mask : pd.Series
 769            Boolean mask identifying rows to apply budgets to. Built by
 770            :meth:`OptionScope.build_mask`.
 771        scope : OptionScope or None, optional
 772            Scope used to resolve per-tool inflation multipliers from
 773            ``media_prices``. When ``None``, only the base price is used.
 774
 775        Returns
 776        -------
 777        pd.DataFrame
 778            Reference to ``ds`` (mutated in place).
 779        """
 780        trans_dict = self.model_rep.trans_dict
 781        if trans_dict is None:
 782            return ds
 783
 784        media_prices = getattr(self.model_rep, "media_prices", {})
 785        if media_prices is None:
 786            media_prices = {}
 787
 788        cols_known = set(ds.columns)
 789
 790        for k, val in budget_items.items():
 791            if k not in trans_dict.get("function", {}):
 792                continue
 793
 794            fun_name = trans_dict["function"][k]
 795            var_name = trans_dict["variable"][k]
 796
 797            if (
 798                fun_name not in locals()
 799                and fun_name not in globals()
 800                and not (
 801                    len(fun_name.split(".")) > 1
 802                    and getattr(
 803                        globals().get(".".join(fun_name.split(".")[:-1]), None),
 804                        fun_name.split(".")[-1],
 805                        False,
 806                    )
 807                )
 808            ):
 809                if fun_name not in _warned_missing_translators:
 810                    _warned_missing_translators.add(fun_name)
 811                    logging.warning(
 812                        f"Translator '{fun_name}' for budget item '{k}' not found "
 813                        f"in node module globals — skipping budget item"
 814                    )
 815                continue
 816
 817            base_price = trans_dict.get("base_price", {}).get(k, 1.0)
 818            if scope is not None:
 819                scope_prices = media_prices.get(scope.scope_name, {})
 820                multiplier = scope_prices.get(k, scope.inflation_multiplier)
 821                final_price = base_price * multiplier
 822            else:
 823                final_price = base_price
 824
 825            ds = eval(fun_name)(model_conf, ds, self.model_rep, k, var_name, val, ds_mask, media_price=final_price)
 826
 827            cols_now = set(ds.columns)
 828            new_cols = cols_now - cols_known
 829            if new_cols:
 830                for col in sorted(new_cols):
 831                    key = (k, col)
 832                    if key not in _warned_new_translation_cols:
 833                        _warned_new_translation_cols.add(key)
 834                        logging.warning(
 835                            f"Translation of budget item '{k}' by translator "
 836                            f"'{fun_name}' created new column '{col}' "
 837                            f"(declared target variable: '{var_name}'). "
 838                            "Translation is expected to overwrite an existing "
 839                            "dataset column — if this new column is not read by "
 840                            "the model, the budget is silently unused."
 841                        )
 842            cols_known = cols_now
 843
 844        return ds

 846    def _translate_long(
 847        self,
 848        ds: pd.DataFrame,
 849        model_conf: ModelConf,
 850        budget_items: dict[str, float],
 851        scope: OptionScope,
 852        allow_past: bool,
 853    ) -> pd.DataFrame:
 854        """
 855        Apply the same budget items beyond the scope end date into the
 856        forecast horizon.
 857
 858        Builds a mask from ``scope.date_end + 1 day`` to the last
 859        future row in the dataset and delegates to
 860        :meth:`_translate_budget`. This enables long-term forecasting
 861        where a one-year budget scenario is projected forward through
 862        all available future periods.
 863
 864        Parameters
 865        ----------
 866        ds : pd.DataFrame
 867            Dataset to modify in place.
 868        model_conf : ModelConf
 869            Model configuration.
 870        budget_items : dict[str, float]
 871            Same budget items used for the exact-period translation.
 872        scope : OptionScope
 873            The scope whose ``date_end`` defines the long-term start.
 874        allow_past : bool
 875            Forwarded to the mask builder. Typically ``False``.
 876
 877        Returns
 878        -------
 879        pd.DataFrame
 880            Reference to ``ds`` (mutated in place).
 881        """
 882        long_start = scope.date_end + pd.DateOffset(1)
 883        long_end = ds.loc[ds["listed"] == 4, "date"].max()
 884        ds_mask = (ds["date"] >= long_start) & (ds["date"] <= long_end)
 885        for col, values in scope.filters.items():
 886            if col in ds.columns:
 887                ds_mask = ds_mask & ds[col].isin(values)
 888        if not allow_past:
 889            ds_mask = ds_mask & ds["listed"].isin([4])
 890        ds = self._translate_budget(ds, model_conf, budget_items, ds_mask, scope)
 891
 892        return ds

 894    def _summarize(
 895        self,
 896        model_conf: ModelConf,
 897        budget_items: dict[str, float],
 898        scope: OptionScope,
 899        dt_pred: pd.DataFrame,
 900    ) -> dict:
 901        """
 902        Derive per-scope summary metrics from a shared prediction.
 903
 904        Computes three slices of ``dt_pred`` by intersecting date ranges
 905        with the scope's entity filter:
 906
 907        - **Exact period**: ``date_start`` -> ``date_end``. KPI aggregated
 908          via ``sum`` or ``mean`` per ``model_conf.summary_type``.
 909        - **Long period**: ``date_start`` -> last future date. Used for
 910          long-term forecast metrics when the scope end is before the
 911          data horizon. Falls back to exact if scope already covers the
 912          full horizon.
 913        - **Reference period**: ``ref_date_start`` -> ``ref_date_end``.
 914          Used to compute growth (forecast / reference).
 915
 916        Both volume (``predicted``) and value (``predicted * price_var``)
 917        variants are produced when a price variable is configured.
 918
 919        Reference aggregates (``pred_ref_*``, ``ref_*``) follow the same
 920        ``summary_type`` aggregation as the exact period, so growth
 921        compares like with like.
 922
 923        Growth is computed as ``pred_exact / ref`` for both volume and
 924        value, with a safety floor of ``1e-10`` to avoid division by zero.
 925
 926        If ``model_rep.option_summary_adhoc`` is set, the summary is
 927        replaced by the adhoc function's return value.
 928
 929        Parameters
 930        ----------
 931        model_conf : ModelConf
 932            Model configuration (provides ``summary_type`` and
 933            ``price_var``).
 934        budget_items : dict[str, float]
 935            Budget items for this scope (passed to adhoc summary).
 936        scope : OptionScope
 937            Scope defining the date ranges and entity filter.
 938        dt_pred : pd.DataFrame
 939            Full prediction DataFrame with columns ``date``, ``listed``,
 940            ``predicted``, ``observed``, and any entity filter columns.
 941
 942        Returns
 943        -------
 944        dict
 945            Summary with keys:
 946            ``pred_exact_vol``, ``pred_exact_val`` (if price var exists),
 947            ``pred_long_vol``, ``pred_long_val`` (if price var exists),
 948            ``pred_ref_vol``, ``pred_ref_val`` (if price var exists),
 949            ``ref_vol``, ``ref_val`` (if price var exists),
 950            ``growth_vol``, ``growth_val`` (if price var exists).
 951        """
 952        summary_type = getattr(model_conf, "summary_type", "sum")
 953        date_max = dt_pred["date"].max()
 954        date_obs_max = dt_pred.loc[dt_pred["listed"].isin([2, 3]), "date"].max()
 955
 956        summary: dict = {}
 957
 958        entity_mask = scope.build_entity_mask(dt_pred)
 959
 960        exact_mask = entity_mask & (
 961            (dt_pred["date"] >= scope.date_start)
 962            & (dt_pred["date"] <= scope.date_end)
 963            & (dt_pred["listed"].isin([2, 3, 4]))
 964        )
 965
 966        if summary_type == "fin":
 967            date_end = dt_pred.loc[dt_pred["listed"].isin([2, 3, 4]), "date"].unique()
 968            dlt = abs(date_end - scope.date_end)
 969            ind = dlt.argmin()
 970            date_end = date_end[ind]
 971            exact_mask = exact_mask & (dt_pred["date"] == date_end)
 972
 973        if summary_type in ("mean", "mean_fin"):
 974            summary["pred_exact_vol"] = dt_pred.loc[exact_mask, "predicted"].mean()
 975        else:
 976            summary["pred_exact_vol"] = dt_pred.loc[exact_mask, "predicted"].sum()
 977
 978        price_var = _get_price_var(model_conf)
 979        if price_var is not None:
 980            price_val = dt_pred.loc[exact_mask, "predicted"] * dt_pred.loc[exact_mask, model_conf.price_var]
 981            if summary_type in ("mean", "mean_fin"):
 982                summary["pred_exact_val"] = price_val.mean()
 983            else:
 984                summary["pred_exact_val"] = price_val.sum()
 985
 986        if scope.date_end < date_max:
 987            long_mask = entity_mask & (
 988                (dt_pred["date"] >= scope.date_start)
 989                & (dt_pred["listed"].isin([2, 3, 4]))
 990            )
 991            if summary_type in ("mean", "mean_fin"):
 992                summary["pred_long_vol"] = dt_pred.loc[long_mask, "predicted"].mean()
 993            else:
 994                summary["pred_long_vol"] = dt_pred.loc[long_mask, "predicted"].sum()
 995
 996            if price_var is not None:
 997                price_val = dt_pred.loc[long_mask, "predicted"] * dt_pred.loc[long_mask, model_conf.price_var]
 998                if summary_type in ("mean", "mean_fin"):
 999                    summary["pred_long_val"] = price_val.mean()
1000                else:
1001                    summary["pred_long_val"] = price_val.sum()
1002        else:
1003            summary["pred_long_vol"] = summary["pred_exact_vol"]
1004            summary["pred_long_val"] = summary.get("pred_exact_val")
1005
1006        ref_mask = entity_mask & (
1007            (dt_pred["date"] >= scope.ref_date_start)
1008            & (dt_pred["date"] <= scope.ref_date_end)
1009            & (dt_pred["listed"].isin([2, 3, 4]))
1010        )
1011
1012        if summary_type in ("mean", "mean_fin"):
1013            summary["pred_ref_vol"] = dt_pred.loc[ref_mask, "predicted"].mean()
1014        else:
1015            summary["pred_ref_vol"] = dt_pred.loc[ref_mask, "predicted"].sum()
1016
1017        if price_var is not None:
1018            price_val = dt_pred.loc[ref_mask, "predicted"] * dt_pred.loc[ref_mask, model_conf.price_var]
1019            if summary_type in ("mean", "mean_fin"):
1020                summary["pred_ref_val"] = price_val.mean()
1021            else:
1022                summary["pred_ref_val"] = price_val.sum()
1023
1024        if scope.ref_date_end <= date_obs_max:
1025            if summary_type in ("mean", "mean_fin"):
1026                summary["ref_vol"] = dt_pred.loc[ref_mask, "observed"].mean()
1027            else:
1028                summary["ref_vol"] = dt_pred.loc[ref_mask, "observed"].sum()
1029            if price_var is not None:
1030                price_val = dt_pred.loc[ref_mask, "observed"] * dt_pred.loc[ref_mask, model_conf.price_var]
1031                if summary_type in ("mean", "mean_fin"):
1032                    summary["ref_val"] = price_val.mean()
1033                else:
1034                    summary["ref_val"] = price_val.sum()
1035        else:
1036            obs_ref_mask = ref_mask & (dt_pred["date"] <= date_obs_max)
1037            pred_ref_mask = ref_mask & (dt_pred["date"] > date_obs_max)
1038            ref_vol_series = pd.concat([
1039                dt_pred.loc[obs_ref_mask, "observed"],
1040                dt_pred.loc[pred_ref_mask, "predicted"],
1041            ])
1042            if summary_type in ("mean", "mean_fin"):
1043                summary["ref_vol"] = ref_vol_series.mean()
1044            else:
1045                summary["ref_vol"] = ref_vol_series.sum()
1046            if price_var is not None:
1047                ref_val_series = pd.concat([
1048                    dt_pred.loc[obs_ref_mask, "observed"] * dt_pred.loc[obs_ref_mask, model_conf.price_var],
1049                    dt_pred.loc[pred_ref_mask, "predicted"] * dt_pred.loc[pred_ref_mask, model_conf.price_var],
1050                ])
1051                if summary_type in ("mean", "mean_fin"):
1052                    summary["ref_val"] = ref_val_series.mean()
1053                else:
1054                    summary["ref_val"] = ref_val_series.sum()
1055
1056        ref_vol = float(summary.get("ref_vol", 1.0))
1057        summary["growth_vol"] = float(summary["pred_exact_vol"]) / max(ref_vol, 1e-10)
1058
1059        if "pred_exact_val" in summary and "ref_val" in summary:
1060            ref_val = float(summary["ref_val"])
1061            summary["growth_val"] = float(summary["pred_exact_val"]) / max(ref_val, 1e-10)
1062
1063        if (
1064            pd.notna(self.model_rep.option_summary_adhoc)
1065            and self.model_rep.option_summary_adhoc is not None
1066        ):
1067            summary = self.model_rep.option_summary_adhoc(
1068                model_conf, self.model_rep, budget_items, scope, dt_pred
1069            )
1070
1071        return summary

 192                "ModelNode requires a model — pass model= or model_path="

 196            model, loaded_ds, _ = model_load_ext(model_path, missed_stop=missed_stop)
 197            if ds is None:
 198                ds = loaded_ds

 200        if ds is None:
 201            fermatrica_error("ds must be provided")

 203        self.model = model
 204        self.ds = ds.copy()
 205        self.dt_pred = dt_pred
 206        self.scope_names = set(scope_names) if scope_names is not None else None
 207        self.static = static
 208        self._display_only = _display_only
 209        if model_rep is not None:
 210            self.model_rep = model_rep

 212            self.model_rep = ModelRep(ds, **model_rep_kwargs)

 244    def load(cls, path: str, model_rep_path: str | None = None):
 245        """
 246        Load a ModelNode from a saved model directory on disk.
 247
 248        Reads the model and dataset via ``model_load_ext`` and the
 249        ModelRep from ``model_rep.pkl.lzma`` in the same directory
 250        (or a custom path). ``model_rep.pkl.lzma`` is optional — when
 251        absent (modelling-stage saves) a default ModelRep is created
 252        from the dataset (English, palettes derived from ``ds``) and a
 253        warning is logged.

 256        ----------

 259            (as produced by :meth:`ModelNode.save`).
 260        model_rep_path : str or None, optional
 261            Explicit path to the ``model_rep.pkl.lzma`` file. When
 262            ``None`` (default), looks for ``model_rep.pkl.lzma``
 263            inside ``path``.
 264
 265        Returns
 266        -------
 267        ModelNode

 269        """
 270        model, ds, _ = model_load_ext(path, missed_stop=False)
 271
 272        if model_rep_path is None:
 273            model_rep_path = os.path.join(path, "model_rep.pkl.lzma")
 274        if os.path.exists(model_rep_path):
 275            model_rep = read_pickle_compat(model_rep_path)
 276        else:
 277            from fermatrica_rep.meta_model.meta_model import _auto_model_rep

 280                f"ModelRep not found at {model_rep_path} — using default ModelRep"
 281            )

 285
 286        toml_path = os.path.join(path, "meta_conf.toml")

 323    def compute(
 324        self,
 325        option=None,
 326        scopes: list[OptionScope] | None = None,
 327        allow_past: bool = False,
 328        if_exact: bool = False,
 329        *,
 330        return_full: bool = False,
 331    ) -> pd.DataFrame | dict[str, dict | tuple[pd.DataFrame, pd.DataFrame, dict]]:
 332        """
 333        Refresh the model fit, or translate budgets and derive summaries.
 334
 335        When ``option`` is ``None`` (default), runs a single
 336        transform + predict on ``self.ds`` with ``set_start=False``
 337        (refresh — the initial ``transform(set_start=True)`` bootstrap
 338        remains a core modelling step) and stores the results on the node.
 339        ``scopes`` / ``return_full`` are ignored on this path.
 340
 341        With an ``option``, translates budgets into media variables for all
 342        scopes, then runs a single transform + predict and derives per-scope
 343        summaries. Convenience method that chains :meth:`translate` and
 344        :meth:`compute_after_translate`. When used directly on a single
 345        ``ModelNode``, scopes are translated cumulatively — each scope inherits
 346        the previous scope's media values and adstock. This is the standard
 347        entry point; call the two sub-methods separately only when the caller
 348        needs to interleave custom operations between translation and compute.
 349
 350        ``self.ds`` and ``self.dt_pred`` are always updated after the
 351        transform + predict, regardless of ``return_full``.
 352
 353        Args:
 354            option: Option object with budget allocations keyed by scope name.
 355                Must provide ``iter_scopes()`` yielding ``(scope_name,
 356                budget_items)`` tuples. ``None`` runs the no-arg refresh path.
 357            scopes: List of ``OptionScope`` objects defining date ranges,
 358                entity filters, and other scope parameters. Scope names are
 359                matched against ``option.iter_scopes()`` during translation.
 360            allow_past: If ``True``, the scope mask built by
 361                ``OptionScope.build_mask`` includes historical periods
 362                (``listed`` ≤ 3). Default ``False`` restricts to ``listed ==
 363                4`` only.
 364            if_exact: If ``True``, skip long-term translation beyond the scope
 365                end date. Use for exact-period scenarios where post-scope media
 366                should not be projected forward.
 367            return_full: If ``True``, each scope entry is a 3-tuple
 368                ``(ds, dt_pred, summary)``. If ``False`` (default), each
 369                scope entry is just the ``summary`` dict. Follows the same
 370                convention as ``fermatrica.fit_predict``.
 371
 372        Returns:
 373            When ``option`` is ``None``: the ``dt_pred`` DataFrame produced by
 374            the refresh (also stored on ``self.dt_pred``).
 375
 376            With an ``option``: dict mapping ``{scope_name: summary}`` when
 377            ``return_full=False`` (default), or
 378            ``{scope_name: (ds, dt_pred, summary)}`` when ``return_full=True``.
 379
 380            ``ds`` is the transformed dataset (a copy made after
 381            transform/predict — safe to inspect without mutating
 382            ``self.ds``). ``dt_pred`` is the full prediction DataFrame
 383            (shared across all scopes — only the per-scope summary slices
 384            differ). ``summary`` is a dict with keys such as
 385            ``pred_exact_vol``, ``pred_long_vol``, ``growth_vol``, etc.
 386        """
 387        if option is None:
 388            return self._refresh()
 389        self.translate(option, scopes, allow_past=allow_past, if_exact=if_exact)
 390        return self.compute_after_translate(option, scopes, return_full=return_full)
 391
 392    def _refresh(self) -> pd.DataFrame:
 393        """
 394        Run transform (``set_start=False``) + predict and store the result.
 395
 396        Uses the node's current ``ds`` as the transform input, then stores
 397        the transformed dataset and prediction back on the node. Returns the
 398        ``dt_pred`` DataFrame.
 399
 400        ``_check_price_var`` / ``_load_adhoc_code`` are idempotent — safe to
 401        re-run even when already executed during :meth:`translate`.
 402        """
 403        if not getattr(self.model.obj, 'models', None) or 'main' not in self.model.obj.models:
 404            fermatrica_error(
 405                "Model is not fitted — run `fm.fit_predict(ds, model, if_fit=True)` "
 406                "before `node.compute()`"
 407            )
 408        self._check_price_var(self.model.conf, self.ds)
 409        self._load_adhoc_code()
 410        model, ds = transform(ds=self.ds, model=self.model, set_start=False, if_by_ref=False)
 411        dt_pred = predict_ext(model, ds)
 412        self.ds = ds
 413        self.dt_pred = dt_pred
 414        return dt_pred

 416    def translate(
 417        self,
 418        option,
 419        scopes: list[OptionScope] | None = None,
 420        allow_past: bool = False,
 421        if_exact: bool = False,
 422    ):
 423        """
 424        Apply per-scope budget translations and media seasonality into
 425        ``self.ds``.
 426
 427        Iterates scopes sorted by ``OptionScope.date_start`` (not by the
 428        order returned by ``option.iter_scopes()``), applying each scope's
 429        seasonality correction and translating budget values into media
 430        variables directly on ``self.ds``. No copies are made between scopes
 431        — later scopes **cumulatively inherit** the translated media values
 432        and adstock carryover from earlier scopes.
 433
 434        Sorting by start date matters for ``_translate_long``: each scope's
 435        budget is projected forward beyond its ``date_end``, so a scope with
 436        an earlier period would otherwise overwrite the media of a later
 437        scope when its ``_translate_long`` runs after it. Processing scopes
 438        chronologically ensures the latest (target) scope's translation and
 439        long-term projection win.
 440
 441        No transform or predict is performed (fast, mutations only). Designed
 442        as Phase 1 of :meth:`MetaModel.compute` so predecessor ``dt_pred``
 443        merges happen before any heavy computation.
 444
 445        Raises ``FermatricaError`` at computation time when the option
 446        carries budgets for the node's scopes but ``model_rep.trans_dict``
 447        is not set — budgets would otherwise silently remain untranslated.
 448        Empty budget items are allowed without a ``trans_dict``.
 449
 450        Args:
 451            option: Option object with budget allocations keyed by scope name.
 452            scopes: List of ``OptionScope`` objects. Each scope is looked up
 453                by name; unmatched scope entries in the option are silently
 454                skipped.
 455            allow_past: Forwarded to ``OptionScope.build_mask``. When
 456                ``False`` (default), only future rows (``listed == 4``)
 457                receive translated values.
 458            if_exact: If ``True``, skip :meth:`_translate_long` — budget
 459                values are applied only to the exact scope period, not
 460                projected forward.
 461
 462        For a static node this is a no-op: its model has no media RHS, so
 463        budgets and seasonality do not apply to it.
 464        """
 465        if self.static:
 466            return
 467        model_conf = self.model.conf
 468        self._check_price_var(model_conf, self.ds)
 469        self._load_adhoc_code()
 470
 471        bdg_ssn_cols = self._build_bdg_ssn_cols()
 472
 473        # Sort scopes chronologically by their start date so that a scope
 474        # translated later (with a later date_start) is not overwritten by
 475        # _translate_long of an earlier-ending scope processed after it.
 476        scopes_sorted: list[tuple[OptionScope, dict[str, float]]] = []
 477
 478        for scope_name, budget_items in option.iter_scopes():
 479            if self.scope_names is not None and scope_name not in self.scope_names:
 480                continue
 481            scope = self._find_scope(scopes, scope_name)
 482            if scope is None:
 483                logging.warning(
 484                    f"Scope '{scope_name}' not found in provided scopes — skipping"
 485                )
 486                continue
 487            scopes_sorted.append((scope, budget_items))
 488
 489        scopes_sorted.sort(key=lambda x: x[0].date_start)
 490
 491        if self.model_rep.trans_dict is None and any(
 492            budget_items for _, budget_items in scopes_sorted
 493        ):
 494            missing_scopes = [
 495                s.scope_name for s, budget_items in scopes_sorted if budget_items
 496            ]
 497            fermatrica_error(
 498                "No translation dictionary (trans_dict) is set on the node, but "
 499                f"budgets are provided for scope(s): {', '.join(missing_scopes)}. "
 500                "Budgets cannot be translated into media variables. "
 501                "Wire a translation sheet via MetaModel(compute_context=...) "
 502                "or set node.model_rep.trans_dict before compute()."
 503            )
 504
 505        for scope, budget_items in scopes_sorted:
 506            self.ds = self._apply_media_ssn(self.ds, scope, bdg_ssn_cols)
 507            ds_mask = scope.build_mask(self.ds, allow_past=allow_past)
 508            self.ds = self._translate_budget(self.ds, model_conf, budget_items, ds_mask, scope)
 509
 510            if not if_exact:
 511                self.ds = self._translate_long(self.ds, model_conf, budget_items, scope, allow_past)
 512
 513    def compute_after_translate(
 514        self,
 515        option,
 516        scopes: list[OptionScope] | None = None,
 517        *,
 518        return_full: bool = False,
 519    ) -> dict[str, dict | tuple[pd.DataFrame, pd.DataFrame, dict]]:
 520        """
 521        Run a single transform + predict on ``self.ds``, then derive
 522        per-scope summaries from the shared prediction.
 523
 524        Assumes ``self.ds`` has already been mutated by :meth:`translate`
 525        (or equivalent cumulative translation). Runs ``transform`` and
 526        ``predict_ext`` exactly once, then slices the single ``dt_pred``
 527        via each scope's ``date_start`` / ``date_end`` and entity filters
 528        to produce per-scope summaries.
 529
 530        Designed as Phase 2 of :meth:`MetaModel.compute` — called after
 531        all predecessor ``dt_pred`` values have been merged via edge-specific
 532        merge functions.
 533
 534        ``self.ds`` and ``self.dt_pred`` are always updated, regardless of
 535        ``return_full``.
 536
 537        Args:
 538            option: Option object. Only used for ``iter_scopes()`` to
 539                enumerate scope names and budget items for summarization.
 540            scopes: List of ``OptionScope`` objects defining date ranges,
 541                entity filters, and reference periods for summaries.
 542            return_full: If ``True``, each scope entry is a 3-tuple
 543                ``(ds, dt_pred, summary)``. If ``False`` (default), each
 544                scope entry is just the ``summary`` dict.
 545
 546        Returns:
 547            dict mapping ``{scope_name: summary}`` when ``return_full=False``
 548            (default), or ``{scope_name: (ds, dt_pred, summary)}`` when
 549            ``return_full=True``.
 550
 551            ``ds`` is a **copy** of the transform output (safe to inspect).
 552            ``dt_pred`` is the full prediction DataFrame (identical for
 553            every scope — shared reference). ``summary`` is a dict with
 554            keys ``pred_exact_vol``, ``pred_long_vol``,
 555            ``pred_ref_vol``, ``growth_vol``, and their ``_val``
 556            counterparts when a price variable is defined.
 557        """
 558        dt_pred = self._refresh()
 559        ds = self.ds
 560
 561        results = {}
 562        model_conf = self.model.conf
 563        for scope_name, budget_items in option.iter_scopes():
 564            if self.scope_names is not None and scope_name not in self.scope_names:
 565                continue
 566            scope = self._find_scope(scopes, scope_name)
 567            if scope:
 568                summary = self._summarize(model_conf, budget_items, scope, dt_pred)
 569                if return_full:
 570                    results[scope_name] = (ds.copy(), dt_pred, summary)
 571                else:
 572                    results[scope_name] = summary
 573
 574        return results

 628    def _apply_media_ssn(
 629        self, ds: pd.DataFrame, scope: OptionScope, bdg_ssn_cols: set[str]
 630    ) -> pd.DataFrame:
 631        """
 632        Apply media seasonality coefficients to the dataset for a given
 633        scope.
 634
 635        Looks up the seasonality table from ``model_rep.ssn_dict`` using
 636        ``scope.seasonality`` as the key (falling back to
 637        ``scope.scope_name``). Calls ``media_ssn_apply`` to multiply
 638        media variables by their seasonal coefficients within the scope's
 639        date range.
 640
 641        Does nothing if no seasonality dictionary is available or if the
 642        lookup key is not found.
 643
 644        Parameters
 645        ----------
 646        ds : pd.DataFrame
 647            Dataset to modify in place. Media columns are multiplied by
 648            seasonal coefficients.
 649        scope : OptionScope
 650            Scope providing the date range and seasonality lookup key.
 651
 652        Returns
 653        -------
 654        pd.DataFrame
 655            Reference to the same ``ds`` object (mutated in place for
 656            performance — the caller receives it for chaining).
 657        """
 658        if (
 659            hasattr(self.model_rep, "ssn_dict")
 660            and pd.notna(self.model_rep.ssn_dict)
 661            and len(self.model_rep.ssn_dict) > 0
 662        ):
 663            from fermatrica_rep.options.define import media_ssn_apply
 664
 665            lookup_key = scope.seasonality if scope.seasonality else scope.scope_name
 666            if lookup_key in self.model_rep.ssn_dict:
 667                dt_ssn = self.model_rep.ssn_dict[lookup_key]
 668                ds = media_ssn_apply(
 669                    ds=ds,
 670                    dt_ssn=dt_ssn,
 671                    date_start=scope.date_start,
 672                    date_end=scope.date_end,
 673                    bdg_ssn_cols=bdg_ssn_cols,
 674                )
 675        return ds

 677    def _build_bdg_ssn_cols(self) -> set[str]:
 678        """
 679        Build a set of bare seasonality column names (without prefix)
 680        that correspond to budget items.
 681
 682        Derived from ``model_rep.trans_dict`` — each key starting with
 683        ``bdg_`` is stripped to its bare name (e.g. ``bdg_tv`` → ``tv``).
 684        """
 685
 686        trans_dict = self.model_rep.trans_dict
 687        if trans_dict is None:
 688            return set()
 689
 690        bdg_cols: set[str] = set()
 691        for k in trans_dict.get("function", {}):
 692            if k.startswith("bdg_"):
 693                bdg_cols.add(k[4:])
 694        return bdg_cols

 696    def _load_adhoc_code(self):
 697        """
 698        Import adhoc user modules registered on
 699        ``model_rep.adhoc_code_src``.
 700
 701        Adhoc modules may define custom translator functions
 702        (``adhoc_*``) or merge functions that are referenced by name in
 703        the translation dictionary or model graph. These must be imported
 704        into the current module's namespace before dispatch.
 705
 706        Idempotency is keyed on the process-local module globals, not on
 707        per-node instance state: ``import_module_from_string`` registers the
 708        full dotted module name as a key in this module's ``globals()``, so
 709        ``k in globals()`` means "already imported in this process". Gating on
 710        the globals keeps the check correct when a node crosses a process
 711        boundary — a spawn Pool worker unpickles the node but starts with a
 712        fresh node.py namespace, so adhoc modules must be re-imported there.
 713
 714        The imported source is this node's pickled snapshot (``adhoc_code_src``
 715        captured at ``ModelRep`` construction), not the live project adhoc
 716        files. Nodes of one MetaModel may have been saved at different
 717        development stages, so the same adhoc module can be imported from a
 718        different (older) source than the current ``code_py/adhoc/`` code —
 719        check that saved sources are up to date in all nodes before relying on
 720        a loaded model.
 721
 722        Does nothing if ``adhoc_code_src`` is empty or unset.
 723        """
 724        if (
 725            pd.notna(self.model_rep.adhoc_code_src)
 726            and len(self.model_rep.adhoc_code_src) > 0
 727        ):
 728            for k, v in self.model_rep.adhoc_code_src.items():
 729                if k in globals():
 730                    continue
 731                fr_cur_name = inspect.currentframe().f_globals["__name__"]
 732                import_module_from_string(k, v, fr_cur_name)

 734    def _translate_budget(
 735        self,
 736        ds: pd.DataFrame,
 737        model_conf: ModelConf,
 738        budget_items: dict[str, float],
 739        ds_mask: pd.Series,
 740        scope: OptionScope | None = None,
 741    ) -> pd.DataFrame:
 742        """
 743        Dispatch budget items to their registered translator functions.
 744
 745        For each budget item (e.g. ``"bdg_tv": 50``):
 746
 747        1. Looks up the translator function name and target variable
 748           from ``model_rep.trans_dict``.
 749        2. Resolves the effective media price by combining the base
 750           price from ``trans_dict`` with per-scope or per-tool
 751           inflation multipliers.
 752        3. Calls the translator function via ``eval()``, passing the
 753           computed ``media_price`` and the scope mask.
 754
 755        Budget items not registered in ``trans_dict["function"]`` are skipped.
 756        Items whose translator functions cannot be resolved are skipped with a
 757        one-time warning (per unique function name, per process).
 758
 759        Parameters
 760        ----------
 761        ds : pd.DataFrame
 762            Dataset to modify in place. Media columns receive budget
 763            values on rows where ``ds_mask`` is ``True``.
 764        model_conf : ModelConf

 822                final_price = base_price * multiplier
 823            else:

 825

 832                    key = (k, col)
 833                    if key not in _warned_new_translation_cols:
 834                        _warned_new_translation_cols.add(key)
 835                        logging.warning(
 836                            f"Translation of budget item '{k}' by translator "
 837                            f"'{fun_name}' created new column '{col}' "
 838                            f"(declared target variable: '{var_name}'). "
 839                            "Translation is expected to overwrite an existing "
 840                            "dataset column — if this new column is not read by "
 841                            "the model, the budget is silently unused."
 842                        )
 843            cols_known = cols_now
 844
 845        return ds

418        option,

420        allow_past: bool = False,
421        if_exact: bool = False,
422    ):
423        """
424        Apply per-scope budget translations and media seasonality into
425        ``self.ds``.
426
427        Iterates scopes sorted by ``OptionScope.date_start`` (not by the
428        order returned by ``option.iter_scopes()``), applying each scope's
429        seasonality correction and translating budget values into media
430        variables directly on ``self.ds``. No copies are made between scopes
431        — later scopes **cumulatively inherit** the translated media values
432        and adstock carryover from earlier scopes.
433
434        Sorting by start date matters for ``_translate_long``: each scope's
435        budget is projected forward beyond its ``date_end``, so a scope with
436        an earlier period would otherwise overwrite the media of a later
437        scope when its ``_translate_long`` runs after it. Processing scopes
438        chronologically ensures the latest (target) scope's translation and
439        long-term projection win.
440
441        No transform or predict is performed (fast, mutations only). Designed
442        as Phase 1 of :meth:`MetaModel.compute` so predecessor ``dt_pred``
443        merges happen before any heavy computation.
444
445        Raises ``FermatricaError`` at computation time when the option
446        carries budgets for the node's scopes but ``model_rep.trans_dict``
447        is not set — budgets would otherwise silently remain untranslated.
448        Empty budget items are allowed without a ``trans_dict``.
449
450        Args:
451            option: Option object with budget allocations keyed by scope name.
452            scopes: List of ``OptionScope`` objects. Each scope is looked up
453                by name; unmatched scope entries in the option are silently
454                skipped.
455            allow_past: Forwarded to ``OptionScope.build_mask``. When
456                ``False`` (default), only future rows (``listed == 4``)
457                receive translated values.
458            if_exact: If ``True``, skip :meth:`_translate_long` — budget
459                values are applied only to the exact scope period, not
460                projected forward.
461
462        For a static node this is a no-op: its model has no media RHS, so
463        budgets and seasonality do not apply to it.
464        """
465        if self.static:
466            return
467        model_conf = self.model.conf
468        self._check_price_var(model_conf, self.ds)
469        self._load_adhoc_code()
470
471        bdg_ssn_cols = self._build_bdg_ssn_cols()
472
473        # Sort scopes chronologically by their start date so that a scope
474        # translated later (with a later date_start) is not overwritten by
475        # _translate_long of an earlier-ending scope processed after it.
476        scopes_sorted: list[tuple[OptionScope, dict[str, float]]] = []
477
478        for scope_name, budget_items in option.iter_scopes():
479            if self.scope_names is not None and scope_name not in self.scope_names:
480                continue
481            scope = self._find_scope(scopes, scope_name)
482            if scope is None:
483                logging.warning(
484                    f"Scope '{scope_name}' not found in provided scopes — skipping"
485                )
486                continue
487            scopes_sorted.append((scope, budget_items))
488
489        scopes_sorted.sort(key=lambda x: x[0].date_start)
490
491        if self.model_rep.trans_dict is None and any(
492            budget_items for _, budget_items in scopes_sorted
493        ):
494            missing_scopes = [
495                s.scope_name for s, budget_items in scopes_sorted if budget_items
496            ]
497            fermatrica_error(
498                "No translation dictionary (trans_dict) is set on the node, but "
499                f"budgets are provided for scope(s): {', '.join(missing_scopes)}. "
500                "Budgets cannot be translated into media variables. "
501                "Wire a translation sheet via MetaModel(compute_context=...) "
502                "or set node.model_rep.trans_dict before compute()."
503            )
504
505        for scope, budget_items in scopes_sorted:
506            self.ds = self._apply_media_ssn(self.ds, scope, bdg_ssn_cols)
507            ds_mask = scope.build_mask(self.ds, allow_past=allow_past)
508            self.ds = self._translate_budget(self.ds, model_conf, budget_items, ds_mask, scope)
509
510            if not if_exact:
511                self.ds = self._translate_long(self.ds, model_conf, budget_items, scope, allow_past)
