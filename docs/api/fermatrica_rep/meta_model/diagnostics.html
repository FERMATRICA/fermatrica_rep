MetaModelDiagnostics — validates a MetaModel object on demand.

Checks:

  1. Node types — all nodes in the DAG are ModelNode instances
  2. Node attributes — critical attributes are present and valid

  44    @property
  45    def passed(self) -> bool:
  46        return all(r.passed for r in self.results if r.level == "error")
  47
  48    @property
  49    def errors(self) -> list[CheckResult]:
  50        return [r for r in self.results if r.level == "error"]
  51
  52    @property
  53    def warnings(self) -> list[CheckResult]:
  54        return [r for r in self.results if r.level == "warning"]
  55
  56    @property
  57    def infos(self) -> list[CheckResult]:
  58        return [r for r in self.results if r.level == "info"]
  59
  60    @property
  61    def errors_summary(self) -> str:
  62        error_list = self.errors
  63        if not error_list:
  64            return ""
  65
  66        lines = [f"  [FAIL] {r.check_name}: {r.message}" for r in error_list]
  67        lines.append("")
  68        lines.append(f"  Errors: {len(error_list)}")
  69        return "\n".join(lines)
  70
  71    @property
  72    def warnings_summary(self) -> str:
  73        warn_list = self.warnings
  74        if not warn_list:
  75            return ""
  76
  77        lines = [f"  [WARN] {r.check_name}: {r.message}" for r in warn_list]
  78        lines.append("")
  79        lines.append(f"  Warnings: {len(warn_list)}")
  80        return "\n".join(lines)
  81
  82    @property
  83    def summary(self) -> str:
  84        errors = [r for r in self.results if r.level == "error"]
  85        warnings = [r for r in self.results if r.level == "warning"]
  86        infos = [r for r in self.results if r.level == "info"]
  87
  88        lines = []
  89        for r in self.results:
  90            if r.passed:
  91                lines.append(f"  [PASS] {r.check_name}: {r.message}")
  92            elif r.level == "warning":
  93                lines.append(f"  [WARN] {r.check_name}: {r.message}")
  94            else:
  95                lines.append(f"  [FAIL] {r.check_name}: {r.message}")
  96
  97        lines.append("")
  98        skipped_str = ", ".join(self.checks_skipped) if self.checks_skipped else "none"
  99        lines.append(f"  Errors: {len(errors)}, Warnings: {len(warnings)}, Info: {len(infos)}")
 100        lines.append(f"  Skipped: {skipped_str}")
 101
 102        return "\n".join(lines)

 110    def run(
 111        self,
 112        options: list | dict | None = None,
 113        scopes: list[OptionScope] | None = None,
 114        raise_on_first: bool = False,
 115    ) -> DiagnosticsReport:
 116        """
 117        Run all diagnostic checks and collect the results.
 118
 119        ``options`` accepts either a ``list`` of ``Option`` objects or a
 120        ``dict`` (e.g. ``ctx.options``) — dict values are used in
 121        insertion order.
 122        """
 123        if isinstance(options, dict):
 124            options = list(options.values())
 125
 126        report = DiagnosticsReport(results=[], checks_ran=[], checks_skipped=[])
 127
 128        self._check_node_types(report, raise_on_first)
 129        self._check_node_attrs(report, raise_on_first)
 130        self._check_edge_resolution(report, raise_on_first)
 131
 132        if options is not None and scopes is not None:
 133            if len(options) >= 2:
 134                self._check_merge_effects(report, options, scopes, raise_on_first)
 135            else:
 136                report.checks_skipped.append(
 137                    "merge_effect: need at least 2 options (got %d)" % len(options)
 138                )
 139        else:
 140            missing = []
 141            if options is None:
 142                missing.append("options")
 143            if scopes is None:
 144                missing.append("scopes")
 145            report.checks_skipped.append("merge_effect: missing %s" % ", ".join(missing))
 146
 147        return report

 572    def _resolve_merge_fn(self, merge_fn: str):
 573        caller_locals = locals()
 574        caller_globals = globals()
 575
 576        if merge_fn in caller_locals or merge_fn in caller_globals:
 577            try:
 578                return eval(merge_fn)
 579            except Exception:
 580                pass
 581
 582        if len(merge_fn.split(".")) > 1:
 583            mod_name = ".".join(merge_fn.split(".")[:-1])
 584            attr_name = merge_fn.split(".")[-1]
 585            mod = caller_globals.get(mod_name, None)
 586            if mod is not None and hasattr(mod, attr_name) and callable(getattr(mod, attr_name)):
 587                try:
 588                    return eval(merge_fn)
 589                except Exception:
 590                    pass
 591
 592        for v in caller_globals.values():
 593            if (
 594                inspect.ismodule(v)
 595                and hasattr(v, merge_fn)
 596                and callable(getattr(v, merge_fn))
 597            ):
 598                return getattr(v, merge_fn)
 599
 600        # Adhoc merge functions are attached to the meta_model module namespace
 601        # by MetaModel._load_adhoc_from_node during compute — search it so
 602        # string merges loaded from meta_conf.toml resolve like they do at
 603        # compute time (edge_resolution and the per-edge probe agree).
 604        mm_module = sys.modules.get("fermatrica_rep.meta_model.meta_model")
 605        if mm_module is not None:
 606            resolved = self._resolve_merge_fn_in_namespace(mm_module.__dict__, merge_fn)
 607            if resolved is not None:
 608                return resolved
 609
 610        return None

 612    @staticmethod
 613    def _resolve_merge_fn_in_namespace(namespace: dict, merge_fn: str):
 614        """
 615        Resolve ``merge_fn`` inside a module namespace dict.
 616
 617        Supports bare names, dotted module attributes (``module.fn``), and a
 618        scan over module values for an attribute matching the name. Returns
 619        ``None`` when not found.
 620        """
 621        if merge_fn in namespace and callable(namespace[merge_fn]):
 622            return namespace[merge_fn]
 623
 624        if len(merge_fn.split(".")) > 1:
 625            mod_name = ".".join(merge_fn.split(".")[:-1])
 626            attr_name = merge_fn.split(".")[-1]
 627            mod = namespace.get(mod_name, None)
 628            if mod is not None and callable(getattr(mod, attr_name, None)):
 629                return getattr(mod, attr_name)
 630
 631        for v in namespace.values():
 632            if inspect.ismodule(v) and callable(getattr(v, merge_fn, None)):
 633                return getattr(v, merge_fn)
 634
 635        return None

 637    def _check_merge_fn_signature(
 638        self,
 639        report: DiagnosticsReport,
 640        left: str,
 641        right: str,
 642        fn,
 643        raise_on_first: bool,
 644    ):
 645        try:
 646            sig = inspect.signature(fn)
 647        except ValueError:
 648            self._add_result(
 649                report,
 650                CheckResult(
 651                    check_name="edge_resolution",
 652                    level="warning",
 653                    passed=True,
 654                    message=f"edge {left} -> {right}: cannot inspect signature of merge fn '{getattr(fn, '__name__', fn)}' (C extension?)",
 655                    details={"left": left, "right": right, "merge_fn": getattr(fn, "__name__", str(fn))},
 656                ),
 657                raise_on_first,
 658            )
 659            return
 660
 661        params = sig.parameters
 662        required = {"ds_cur", "dt_pred_prev"}
 663        missing = required - set(params.keys())
 664
 665        if missing:
 666            self._add_result(
 667                report,
 668                CheckResult(
 669                    check_name="edge_resolution",
 670                    level="warning",
 671                    passed=True,
 672                    message=f"edge {left} -> {right}: merge fn '{getattr(fn, '__name__', fn)}' does not accept required kwargs: {', '.join(sorted(missing))}",
 673                    details={"left": left, "right": right, "merge_fn": getattr(fn, "__name__", str(fn)), "missing_params": sorted(missing)},

 678    def _check_merge_effects(
 679        self,
 680        report: DiagnosticsReport,
 681        options: list,
 682        scopes: list[OptionScope],
 683        raise_on_first: bool,
 684    ):
 685        if options[0].budget_data == options[1].budget_data:
 686            report.checks_skipped.append(
 687                "merge_effect: options '%s' and '%s' are identical"
 688                % (options[0].name, options[1].name)
 689            )
 690            return
 691
 692        saved = {}
 693        for name, node in self.meta_model.nodes.items():
 694            saved[name] = (node.ds.copy(), node.dt_pred)
 695
 696        try:
 697            self.meta_model.compute(options[0], scopes)
 698            ds_A = {name: node.ds.copy() for name, node in self.meta_model.nodes.items()}
 699            dt_pred_A = {
 700                name: node.dt_pred.copy() if node.dt_pred is not None else None
 701                for name, node in self.meta_model.nodes.items()
 702            }
 703
 704            for name, node in self.meta_model.nodes.items():
 705                node.ds, node.dt_pred = saved[name]
 706
 707            self.meta_model.compute(options[1], scopes)
 708            ds_B = {name: node.ds.copy() for name, node in self.meta_model.nodes.items()}
 709            dt_pred_B = {
 710                name: node.dt_pred.copy() if node.dt_pred is not None else None
 711                for name, node in self.meta_model.nodes.items()
 712            }
 713
 714            for name, node in self.meta_model.nodes.items():
 715                node.ds, node.dt_pred = saved[name]
 716
 717            option_influenced = self._option_influenced_nodes()
 718            for name in self.meta_model.nodes:
 719                self._compare_dataframes(
 720                    report, name, "ds", ds_A[name], ds_B[name], scopes, raise_on_first,
 721                    expected_invariant=name not in option_influenced,
 722                )
 723                self._compare_dataframes(
 724                    report, name, "dt_pred", dt_pred_A[name], dt_pred_B[name], scopes, raise_on_first,
 725                    expected_invariant=name not in option_influenced,
 726                )
 727
 728            self._check_edge_propagation(
 729                report, ds_A, dt_pred_A, dt_pred_B, raise_on_first
 730            )
 731
 732        except Exception:
 733            for name, node in self.meta_model.nodes.items():
 734                node.ds, node.dt_pred = saved[name]
 735            raise

 737    def _option_influenced_nodes(self) -> set[str]:
 738        """Nodes whose output may vary between options.
 739
 740        A node is option-influenced when it is non-static (it translates
 741        budgets) or is reachable from a non-static node via edges (it
 742        consumes dynamic upstream predictions). Static nodes outside this
 743        set are expected to be option-invariant.
 744        """
 745        influenced = {
 746            name for name, node in self.meta_model.nodes.items() if not node.static
 747        }
 748        changed = True
 749        while changed:
 750            changed = False
 751            for left, right, _ in self.meta_model.edges:
 752                if left in influenced and right not in influenced:
 753                    influenced.add(right)
 754                    changed = True
 755        return influenced

 757    def _compare_dataframes(
 758        self,
 759        report: DiagnosticsReport,
 760        node_name: str,
 761        df_label: str,
 762        df_A: pd.DataFrame | None,
 763        df_B: pd.DataFrame | None,
 764        scopes: list[OptionScope],
 765        raise_on_first: bool,
 766        expected_invariant: bool = False,
 767    ):
 768        report.checks_ran.append(f"merge_{df_label}")
 769
 770        if df_A is None or df_B is None:
 771            self._add_result(
 772                report,
 773                CheckResult(
 774                    check_name=f"merge_{df_label}",
 775                    level="info",
 776                    passed=True,
 777                    message=f"'{node_name}' {df_label}: one or both DataFrames are None — cannot compare (prediction not computed?)",
 778                    details={"node_name": node_name, "df_label": df_label},
 779                ),
 780                raise_on_first,
 781            )
 782            return
 783
 784        rows_A, cols_A = df_A.shape
 785        rows_B, cols_B = df_B.shape
 786
 787        float_cols = []
 788        other_cols = []
 789        for c in df_A.columns:
 790            if c not in df_B.columns:
 791                continue
 792            try:
 793                if np.issubdtype(df_A[c].dtype, np.floating):
 794                    float_cols.append(c)
 795                else:
 796                    other_cols.append(c)
 797            except TypeError:
 798                other_cols.append(c)
 799
 800        all_identical = True
 801        any_differ = False
 802
 803        if float_cols:
 804            if not np.allclose(
 805                df_A[float_cols].to_numpy(dtype=float, na_value=np.nan),
 806                df_B[float_cols].to_numpy(dtype=float, na_value=np.nan),
 807                rtol=1e-7,
 808                atol=1e-12,
 809                equal_nan=True,
 810            ):
 811                all_identical = False
 812                any_differ = True
 813                diff_cols = []
 814                for c in float_cols:
 815                    if not np.allclose(
 816                        df_A[c].to_numpy(dtype=float, na_value=np.nan),
 817                        df_B[c].to_numpy(dtype=float, na_value=np.nan),
 818                        rtol=1e-7,
 819                        atol=1e-12,
 820                        equal_nan=True,
 821                    ):
 822                        diff_cols.append(c)

 824                    report,

 826                        check_name=f"merge_{df_label}",
 827                        level="info",
 828                        passed=True,
 829                        message=f"'{node_name}' {df_label}: {len(diff_cols)} float columns differ between options: {diff_cols[:10]}",

 839                    diff_other.append(c)

 841            if diff_other:
 842                all_identical = False
 843                any_differ = True
 844                self._add_result(
 845                    report,
 846                    CheckResult(
 847                        check_name=f"merge_{df_label}",
 848                        level="info",
 849                        passed=True,
 850                        message=f"'{node_name}' {df_label}: {len(diff_other)} non-float columns differ between options: {diff_other[:10]}",
 851                        details={"node_name": node_name, "df_label": df_label, "diff_other_cols": diff_other[:10]},
 852                    ),
 853                    raise_on_first,
 854                )
 855

 858                self._add_result(
 859                    report,
 860                    CheckResult(
 861                        check_name=f"merge_{df_label}",
 862                        level="info",
 863                        passed=True,
 864                        message=f"'{node_name}' {df_label}: unchanged between options — static node with no option-influenced upstream (expected invariant)",
 865                        details={"node_name": node_name, "df_label": df_label, "rows": (rows_A, rows_B), "cols": (cols_A, cols_B)},

 867                    raise_on_first,
 868                )

 870                self._add_result(
 871                    report,
 872                    CheckResult(
 873                        check_name=f"merge_{df_label}",
 874                        level="error",
 875                        passed=False,
 876                        message=f"'{node_name}' {df_label}: unchanged between options — merge may not be propagating or options may not affect this node",
 877                        details={"node_name": node_name, "df_label": df_label, "rows": (rows_A, rows_B), "cols": (cols_A, cols_B)},
 878                    ),
 879                    raise_on_first,
 880                )
 881        elif any_differ:
 882            self._add_result(
 883                report,
 884                CheckResult(
 885                    check_name=f"merge_{df_label}",
 886                    level="info",
 887                    passed=True,
 888                    message=f"'{node_name}' {df_label}: differs between options — merge effects propagate",
 889                    details={"node_name": node_name, "df_label": df_label, "rows": (rows_A, rows_B), "cols": (cols_A, cols_B)},

 891                raise_on_first,
 892            )
 893        else:
 894            self._add_result(
 895                report,
 896                CheckResult(
 897                    check_name=f"merge_{df_label}",
 898                    level="info",
 899                    passed=True,
 900                    message=f"'{node_name}' {df_label}: all columns identical between options",

 906    def _check_edge_propagation(
 907        self,
 908        report: DiagnosticsReport,
 909        ds_A: dict,
 910        dt_pred_A: dict,
 911        dt_pred_B: dict,
 912        raise_on_first: bool,
 913    ):
 914        """
 915        Per-edge attribution of the option-driven signal.
 916
 917        The node-level comparison above proves that a node's output *somewhere*
 918        changed between options. In a fan-in DAG (a node with 2+ predecessors)
 919        that can hide a broken edge: one merge propagates the signal while
 920        another silently no-ops or writes constant values, and the node still
 921        differs overall.
 922
 923        For each edge ``left -> right`` with a merge function, this probe
 924        reuses the snapshots from the two option runs: when ``left.dt_pred``
 925        differs between options (upstream signal present), the merge is invoked
 926        directly on a copy of ``right.ds`` with each option's predecessor
 927        prediction. Identical merged output means the edge itself does not
 928        propagate the signal:
 929
 930        - merge writes **nothing** at all → error (silent no-op — the edge is
 931          definitively dead for the active options and the node-level check
 932          cannot see it in a fan-in DAG);
 933        - merge writes columns but produces identical values → warning (a
 934          constant-value merge may be intentional).
 935
 936        ``allow_past=False`` matches the option-path semantics used by the two
 937        :meth:`MetaModel.compute` runs above.
 938        """
 939        for left, right, merge_fn in self.meta_model.edges:
 940            if merge_fn is None:
 941                continue
 942
 943            report.checks_ran.append("merge_edge")
 944
 945            left_dt_A = dt_pred_A.get(left)
 946            left_dt_B = dt_pred_B.get(left)
 947            base = ds_A.get(right)
 948
 949            if left_dt_A is None or left_dt_B is None:
 950                self._add_result(
 951                    report,
 952                    CheckResult(
 953                        check_name="merge_edge",
 954                        level="info",
 955                        passed=True,
 956                        message=f"edge {left} -> {right}: cannot test — left node dt_pred is None in one or both runs",
 957                        details={"left": left, "right": right, "reason": "left dt_pred missing"},
 958                    ),
 959                    raise_on_first,
 960                )
 961                continue
 962
 963            if self._frames_identical(left_dt_A, left_dt_B):
 964                self._add_result(
 965                    report,
 966                    CheckResult(
 967                        check_name="merge_edge",
 968                        level="info",
 969                        passed=True,
 970                        message=f"edge {left} -> {right}: no upstream signal between these options — edge not exercisable",
 971                        details={"left": left, "right": right, "reason": "left dt_pred identical across options"},
 972                    ),
 973                    raise_on_first,
 974                )
 975                continue
 976
 977            fn = self._resolve_merge_fn_for_probe(merge_fn)
 978            if fn is None:
 979                self._add_result(
 980                    report,
 981                    CheckResult(
 982                        check_name="merge_edge",
 983                        level="info",
 984                        passed=True,
 985                        message=f"edge {left} -> {right}: cannot test — merge fn '{merge_fn}' not resolvable (see edge_resolution)",
 986                        details={"left": left, "right": right, "reason": "unresolvable merge fn"},
 987                    ),
 988                    raise_on_first,
 989                )
 990                continue
 991
 992            try:
 993                merged_A = self.meta_model._apply_merge_fn(
 994                    fn, base.copy(), left_dt_A.copy(), allow_past=False
 995                )
 996                merged_B = self.meta_model._apply_merge_fn(
 997                    fn, base.copy(), left_dt_B.copy(), allow_past=False
 998                )
 999            except Exception as exc:
1000                self._add_result(
1001                    report,
1002                    CheckResult(
1003                        check_name="merge_edge",
1004                        level="warning",
1005                        passed=True,
1006                        message=f"edge {left} -> {right}: merge fn raised during propagation probe: {exc}",
1007                        details={"left": left, "right": right, "exception": str(exc)},
1008                    ),
1009                    raise_on_first,
1010                )
1011                continue
1012
1013            diff_left_cols = self._diff_columns(left_dt_A, left_dt_B)
1014            propagates = not self._frames_identical(merged_A, merged_B)
1015            wrote_any = not self._frames_identical(merged_A, base)
1016
1017            if propagates:
1018                self._add_result(
1019                    report,
1020                    CheckResult(
1021                        check_name="merge_edge",
1022                        level="info",
1023                        passed=True,
1024                        message=(
1025                            f"edge {left} -> {right}: merge propagates upstream signal "
1026                            f"(left dt_pred differs in {len(diff_left_cols)} col(s): {diff_left_cols[:5]})"
1027                        ),
1028                        details={"left": left, "right": right, "diff_left_cols": diff_left_cols[:10]},
1029                    ),
1030                    raise_on_first,
1031                )
1032            elif wrote_any:
1033                self._add_result(
1034                    report,
1035                    CheckResult(
1036                        check_name="merge_edge",
1037                        level="warning",
1038                        passed=True,
1039                        message=(
1040                            f"edge {left} -> {right}: merge writes columns but output is identical across options — "
1041                            f"does not propagate upstream signal "
1042                            f"({len(diff_left_cols)} col(s) differ in left dt_pred: {diff_left_cols[:5]})"
1043                        ),
1044                        details={
1045                            "left": left,
1046                            "right": right,
1047                            "diff_left_cols": diff_left_cols[:10],
1048                            "reason": "constant output despite upstream signal",
1049                        },
1050                    ),
1051                    raise_on_first,
1052                )
1053            else:
1054                self._add_result(
1055                    report,
1056                    CheckResult(
1057                        check_name="merge_edge",
1058                        level="error",
1059                        passed=False,
1060                        message=(
1061                            f"edge {left} -> {right}: merge produced no output change on either option — "
1062                            f"silent no-op; upstream signal "
1063                            f"({len(diff_left_cols)} col(s): {diff_left_cols[:5]}) not propagated"
1064                        ),
1065                        details={
1066                            "left": left,
1067                            "right": right,
1068                            "diff_left_cols": diff_left_cols[:10],
1069                            "reason": "no columns written",
1070                        },
1071                    ),
1072                    raise_on_first,
1073                )

1075    def _resolve_merge_fn_for_probe(self, merge_fn) -> "callable | None":
1076        """
1077        Resolve a merge function for the per-edge propagation probe.
1078
1079        Delegates to :meth:`_resolve_merge_fn`, which also searches the
1080        ``fermatrica_rep.meta_model.meta_model`` module namespace where adhoc
1081        merge functions are attached by ``MetaModel._load_adhoc_from_node``
1082        during :meth:`MetaModel.compute`.
1083        """
1084        if callable(merge_fn):
1085            return merge_fn
1086        if not isinstance(merge_fn, str):
1087            return None
1088        return self._resolve_merge_fn(merge_fn)

1090    @staticmethod
1091    def _frames_identical(
1092        df_A: pd.DataFrame | None, df_B: pd.DataFrame | None
1093    ) -> bool:
1094        """Float-aware equality for two DataFrames (same tolerance as the node-level check)."""
1095        if df_A is None or df_B is None:
1096            return df_A is df_B
1097        if df_A.shape != df_B.shape or list(df_A.columns) != list(df_B.columns):
1098            return False
1099
1100        float_cols = []
1101        other_cols = []
1102        for c in df_A.columns:
1103            try:
1104                if np.issubdtype(df_A[c].dtype, np.floating):
1105                    float_cols.append(c)
1106                else:
1107                    other_cols.append(c)
1108            except TypeError:
1109                other_cols.append(c)
1110
1111        if float_cols and not np.allclose(
1112            df_A[float_cols].to_numpy(dtype=float, na_value=np.nan),
1113            df_B[float_cols].to_numpy(dtype=float, na_value=np.nan),
1114            rtol=1e-7,
1115            atol=1e-12,
1116            equal_nan=True,
1117        ):
1118            return False
1119
1120        return all(df_A[c].equals(df_B[c]) for c in other_cols)

1122    @staticmethod
1123    def _diff_columns(df_A: pd.DataFrame, df_B: pd.DataFrame) -> list:
1124        """Return the columns where two DataFrames differ (float-aware)."""
1125        if df_A is None or df_B is None:
1126            return []

1130                diff.append(c)
1131            elif not MetaModelDiagnostics._frames_identical(df_A[[c]], df_B[[c]]):
1132                diff.append(c)
1133        return diff

 120        ``options`` accepts either a ``list`` of ``Option`` objects or a
 121        ``dict`` (e.g. ``ctx.options``) — dict values are used in
 122        insertion order.
 123        """
 124        if isinstance(options, dict):
 125            options = list(options.values())
 126
 127        report = DiagnosticsReport(results=[], checks_ran=[], checks_skipped=[])
 128
 129        self._check_node_types(report, raise_on_first)
 130        self._check_node_attrs(report, raise_on_first)
 131        self._check_edge_resolution(report, raise_on_first)
 132
 133        if options is not None and scopes is not None:
 134            if len(options) >= 2:
 135                self._check_merge_effects(report, options, scopes, raise_on_first)
 136            else:
 137                report.checks_skipped.append(
 138                    "merge_effect: need at least 2 options (got %d)" % len(options)
 139                )
 140        else:
 141            missing = []
 142            if options is None:
 143                missing.append("options")
 144            if scopes is None:
 145                missing.append("scopes")
 146            report.checks_skipped.append("merge_effect: missing %s" % ", ".join(missing))
 147
 148        return report

 573    def _resolve_merge_fn(self, merge_fn: str):
 574        caller_locals = locals()
 575        caller_globals = globals()
 576
 577        if merge_fn in caller_locals or merge_fn in caller_globals:
 578            try:
 579                return eval(merge_fn)
 580            except Exception:
 581                pass
 582
 583        if len(merge_fn.split(".")) > 1:
 584            mod_name = ".".join(merge_fn.split(".")[:-1])
 585            attr_name = merge_fn.split(".")[-1]
 586            mod = caller_globals.get(mod_name, None)
 587            if mod is not None and hasattr(mod, attr_name) and callable(getattr(mod, attr_name)):
 588                try:
 589                    return eval(merge_fn)
 590                except Exception:
 591                    pass
 592
 593        for v in caller_globals.values():
 594            if (
 595                inspect.ismodule(v)
 596                and hasattr(v, merge_fn)
 597                and callable(getattr(v, merge_fn))
 598            ):
 599                return getattr(v, merge_fn)
 600
 601        # Adhoc merge functions are attached to the meta_model module namespace
 602        # by MetaModel._load_adhoc_from_node during compute — search it so
 603        # string merges loaded from meta_conf.toml resolve like they do at
 604        # compute time (edge_resolution and the per-edge probe agree).
 605        mm_module = sys.modules.get("fermatrica_rep.meta_model.meta_model")
 606        if mm_module is not None:
 607            resolved = self._resolve_merge_fn_in_namespace(mm_module.__dict__, merge_fn)
 608            if resolved is not None:
 609                return resolved
 610
 611        return None

 613    @staticmethod
 614    def _resolve_merge_fn_in_namespace(namespace: dict, merge_fn: str):
 615        """
 616        Resolve ``merge_fn`` inside a module namespace dict.
 617
 618        Supports bare names, dotted module attributes (``module.fn``), and a
 619        scan over module values for an attribute matching the name. Returns
 620        ``None`` when not found.
 621        """
 622        if merge_fn in namespace and callable(namespace[merge_fn]):
 623            return namespace[merge_fn]
 624
 625        if len(merge_fn.split(".")) > 1:
 626            mod_name = ".".join(merge_fn.split(".")[:-1])
 627            attr_name = merge_fn.split(".")[-1]
 628            mod = namespace.get(mod_name, None)
 629            if mod is not None and callable(getattr(mod, attr_name, None)):
 630                return getattr(mod, attr_name)
 631
 632        for v in namespace.values():
 633            if inspect.ismodule(v) and callable(getattr(v, merge_fn, None)):
 634                return getattr(v, merge_fn)
 635
 636        return None

 638    def _check_merge_fn_signature(
 639        self,
 640        report: DiagnosticsReport,
 641        left: str,
 642        right: str,
 643        fn,
 644        raise_on_first: bool,
 645    ):
 646        try:
 647            sig = inspect.signature(fn)
 648        except ValueError:
 649            self._add_result(
 650                report,
 651                CheckResult(
 652                    check_name="edge_resolution",
 653                    level="warning",
 654                    passed=True,
 655                    message=f"edge {left} -> {right}: cannot inspect signature of merge fn '{getattr(fn, '__name__', fn)}' (C extension?)",
 656                    details={"left": left, "right": right, "merge_fn": getattr(fn, "__name__", str(fn))},
 657                ),
 658                raise_on_first,
 659            )
 660            return
 661
 662        params = sig.parameters
 663        required = {"ds_cur", "dt_pred_prev"}
 664        missing = required - set(params.keys())

 666        if missing:
 667            self._add_result(
 668                report,
 669                CheckResult(
 670                    check_name="edge_resolution",
 671                    level="warning",
 672                    passed=True,
 673                    message=f"edge {left} -> {right}: merge fn '{getattr(fn, '__name__', fn)}' does not accept required kwargs: {', '.join(sorted(missing))}",
 674                    details={"left": left, "right": right, "merge_fn": getattr(fn, "__name__", str(fn)), "missing_params": sorted(missing)},
 675                ),
 676                raise_on_first,
 677            )

 679    def _check_merge_effects(
 680        self,
 681        report: DiagnosticsReport,
 682        options: list,
 683        scopes: list[OptionScope],
 684        raise_on_first: bool,
 685    ):
 686        if options[0].budget_data == options[1].budget_data:
 687            report.checks_skipped.append(
 688                "merge_effect: options '%s' and '%s' are identical"
 689                % (options[0].name, options[1].name)
 690            )
 691            return
 692
 693        saved = {}
 694        for name, node in self.meta_model.nodes.items():
 695            saved[name] = (node.ds.copy(), node.dt_pred)
 696
 697        try:
 698            self.meta_model.compute(options[0], scopes)
 699            ds_A = {name: node.ds.copy() for name, node in self.meta_model.nodes.items()}
 700            dt_pred_A = {
 701                name: node.dt_pred.copy() if node.dt_pred is not None else None
 702                for name, node in self.meta_model.nodes.items()
 703            }
 704
 705            for name, node in self.meta_model.nodes.items():
 706                node.ds, node.dt_pred = saved[name]
 707
 708            self.meta_model.compute(options[1], scopes)
 709            ds_B = {name: node.ds.copy() for name, node in self.meta_model.nodes.items()}
 710            dt_pred_B = {
 711                name: node.dt_pred.copy() if node.dt_pred is not None else None
 712                for name, node in self.meta_model.nodes.items()
 713            }
 714
 715            for name, node in self.meta_model.nodes.items():
 716                node.ds, node.dt_pred = saved[name]
 717
 718            option_influenced = self._option_influenced_nodes()
 719            for name in self.meta_model.nodes:
 720                self._compare_dataframes(
 721                    report, name, "ds", ds_A[name], ds_B[name], scopes, raise_on_first,
 722                    expected_invariant=name not in option_influenced,
 723                )
 724                self._compare_dataframes(
 725                    report, name, "dt_pred", dt_pred_A[name], dt_pred_B[name], scopes, raise_on_first,
 726                    expected_invariant=name not in option_influenced,
 727                )
 728
 729            self._check_edge_propagation(
 730                report, ds_A, dt_pred_A, dt_pred_B, raise_on_first
 731            )
 732
 733        except Exception:
 734            for name, node in self.meta_model.nodes.items():
 735                node.ds, node.dt_pred = saved[name]
 736            raise

 738    def _option_influenced_nodes(self) -> set[str]:
 739        """Nodes whose output may vary between options.
 740
 741        A node is option-influenced when it is non-static (it translates
 742        budgets) or is reachable from a non-static node via edges (it
 743        consumes dynamic upstream predictions). Static nodes outside this
 744        set are expected to be option-invariant.
 745        """
 746        influenced = {
 747            name for name, node in self.meta_model.nodes.items() if not node.static
 748        }
 749        changed = True
 750        while changed:
 751            changed = False
 752            for left, right, _ in self.meta_model.edges:
 753                if left in influenced and right not in influenced:
 754                    influenced.add(right)
 755                    changed = True
 756        return influenced

 758    def _compare_dataframes(
 759        self,
 760        report: DiagnosticsReport,
 761        node_name: str,
 762        df_label: str,
 763        df_A: pd.DataFrame | None,
 764        df_B: pd.DataFrame | None,
 765        scopes: list[OptionScope],
 766        raise_on_first: bool,
 767        expected_invariant: bool = False,
 768    ):
 769        report.checks_ran.append(f"merge_{df_label}")
 770
 771        if df_A is None or df_B is None:
 772            self._add_result(
 773                report,
 774                CheckResult(
 775                    check_name=f"merge_{df_label}",
 776                    level="info",
 777                    passed=True,
 778                    message=f"'{node_name}' {df_label}: one or both DataFrames are None — cannot compare (prediction not computed?)",
 779                    details={"node_name": node_name, "df_label": df_label},
 780                ),
 781                raise_on_first,
 782            )
 783            return
 784
 785        rows_A, cols_A = df_A.shape
 786        rows_B, cols_B = df_B.shape
 787
 788        float_cols = []
 789        other_cols = []
 790        for c in df_A.columns:
 791            if c not in df_B.columns:
 792                continue
 793            try:
 794                if np.issubdtype(df_A[c].dtype, np.floating):
 795                    float_cols.append(c)
 796                else:
 797                    other_cols.append(c)
 798            except TypeError:
 799                other_cols.append(c)
 800
 801        all_identical = True

 907    def _check_edge_propagation(
 908        self,
 909        report: DiagnosticsReport,
 910        ds_A: dict,
 911        dt_pred_A: dict,
 912        dt_pred_B: dict,
 913        raise_on_first: bool,
 914    ):
 915        """
 916        Per-edge attribution of the option-driven signal.
 917
 918        The node-level comparison above proves that a node's output *somewhere*
 919        changed between options. In a fan-in DAG (a node with 2+ predecessors)
 920        that can hide a broken edge: one merge propagates the signal while
 921        another silently no-ops or writes constant values, and the node still
 922        differs overall.
 923
 924        For each edge ``left -> right`` with a merge function, this probe
 925        reuses the snapshots from the two option runs: when ``left.dt_pred``
 926        differs between options (upstream signal present), the merge is invoked
 927        directly on a copy of ``right.ds`` with each option's predecessor
 928        prediction. Identical merged output means the edge itself does not
 929        propagate the signal:
 930
 931        - merge writes **nothing** at all → error (silent no-op — the edge is
 932          definitively dead for the active options and the node-level check
 933          cannot see it in a fan-in DAG);
 934        - merge writes columns but produces identical values → warning (a
 935          constant-value merge may be intentional).
 936
 937        ``allow_past=False`` matches the option-path semantics used by the two
 938        :meth:`MetaModel.compute` runs above.
 939        """
 940        for left, right, merge_fn in self.meta_model.edges:
 941            if merge_fn is None:
 942                continue

 945

 947            left_dt_B = dt_pred_B.get(left)

 961                )

 969                        level="info",
 970                        passed=True,
 971                        message=f"edge {left} -> {right}: no upstream signal between these options — edge not exercisable",
 972                        details={"left": left, "right": right, "reason": "left dt_pred identical across options"},
