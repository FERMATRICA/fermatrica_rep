  56        compute_context: "ComputeContext | str | None" = None,

  58        """
  59        Initialise a MetaModel.
  60
  61        ``nodes`` accepts either a ``dict[str, ModelNode]`` (multi-node)
  62        or a single ``ModelNode`` (auto-named ``"model"``). Pass
  63        ``edges`` when multiple nodes form a DAG with dependencies.
  64
  65        When ``compute_context`` is provided (an :class:`ComputeContext`
  66        or a path to a context XLSX file), each node's ``ModelRep`` is
  67        wired at construction time: ``trans_dict`` (per-node
  68        ``translation_<node>`` sheet, or the single shared
  69        ``translation`` sheet when exactly one exists), ``ssn_dict`` and
  70        ``media_prices`` are assigned only where the node does not already
  71        carry an explicit value. Scopes are **not** stored on the
  72        MetaModel — pass them to :meth:`compute` as the ``scopes``
  73        argument (e.g. ``ctx.scopes``) so different scope sets can be
  74        applied to the same model.
  75
  76        Parameters
  77        ----------
  78        nodes : dict[str, ModelNode] or ModelNode
  79            Node(s) to include. A single ``ModelNode`` is wrapped as
  80            ``{"model": node}`` automatically.
  81        edges : list[tuple[str, str, Callable or str or None]] or None, optional
  82            Dependency edges as ``(left, right, merge_fn)`` tuples.
  83            ``merge_fn`` is a callable, a string name resolved at
  84            compute time, or ``None`` for parallel independent branches.
  85        compute_context : ComputeContext or str or None, optional
  86            Pre-built :class:`ComputeContext` or a path to a context
  87            XLSX file (loaded via :func:`fermatrica_rep.options.context.load_context`).
  88        """
  89        if isinstance(nodes, ModelNode):
  90            nodes = {"model": nodes}
  91        self.nodes = nodes
  92        self.edges = edges if edges is not None else []
  93        if compute_context is not None:
  94            from fermatrica_rep.options.context import load_context
  95
  96            if isinstance(compute_context, str):
  97                compute_context = load_context(compute_context)
  98
  99            for node_name, node in self.nodes.items():
 100                mr = node.model_rep
 101                if mr.trans_dict is None and not node.static:
 102                    trans_dict = compute_context.trans_dicts.get(node_name)
 103                    if trans_dict is None and len(compute_context.trans_dicts) == 1:
 104                        trans_dict = next(iter(compute_context.trans_dicts.values()))
 105                    if trans_dict is None and len(compute_context.trans_dicts) > 1:
 106                        fermatrica_error(
 107                            f"Node '{node_name}' has no matching translation sheet in "
 108                            f"{compute_context.path}. Found sheets: "
 109                            f"{sorted(compute_context.trans_dicts.keys())}. "
 110                            "Name the node to match a 'translation_<node>' sheet "
 111                            "or pass trans_dict explicitly."
 112                        )
 113                    mr.trans_dict = trans_dict
 114                if mr.ssn_dict is None:
 115                    mr.ssn_dict = compute_context.ssn_dict
 116                if getattr(mr, "media_prices", None) is None:
 117                    mr.media_prices = compute_context.media_prices
 118        self._validate()
 119        self._init_finish()

 125    def __str__(self) -> str:
 126        lines = [f"MetaModel: {len(self.nodes)} nodes, {len(self.edges)} edges"]
 127        if self.edges:
 128            lines.append("  Graph:")
 129            for left, right, merge_fn in self.edges:
 130                edge_str = f"    {left} -> {right}"
 131                if merge_fn is not None:
 132                    label = merge_fn if isinstance(merge_fn, str) else getattr(merge_fn, '__name__', merge_fn)
 133                    edge_str += f"  [{label}]"
 134                lines.append(edge_str)
 135        else:
 136            lines.append("  (single node, no edges)")
 137        order = self._topological_order()
 138        lines.append(f"  Order: {' -> '.join(order)}")
 139        return "\n".join(lines)

 141    def _validate(self):
 142        """
 143        Validate node and edge consistency.
 144
 145        Raises FermatricaError if the graph is malformed.
 146        """
 147        node_names = set(self.nodes.keys())
 148        n_nodes = len(node_names)
 149
 150        if n_nodes == 0:
 151            fermatrica_error("MetaModel must contain at least one node.")
 152
 153        if n_nodes > 1 and not self.edges:
 154            fermatrica_error(
 155                f"MetaModel has {n_nodes} nodes but no edges provided. "
 156                "Edges are required when more than one node is present."
 157            )
 158
 159        if not self.edges:
 160            return
 161
 162        for left, right, _ in self.edges:
 163            if left not in node_names:
 164                fermatrica_error(
 165                    f"Edge left node '{left}' is not in MetaModel nodes: {sorted(node_names)}"
 166                )
 167            if right not in node_names:
 168                fermatrica_error(
 169                    f"Edge right node '{right}' is not in MetaModel nodes: {sorted(node_names)}"
 170                )
 171            if left == right:
 172                fermatrica_error(f"Self-loop detected: '{left}' -> '{right}'.")
 173
 174        seen_edges = set()
 175        for edge in self.edges:
 176            key = (edge[0], edge[1])
 177            if key in seen_edges:
 178                fermatrica_error(f"Duplicate edge: '{key[0]}' -> '{key[1]}'.")
 179            seen_edges.add(key)
 180
 181        referenced = set()
 182        for left, right, _ in self.edges:
 183            referenced.add(left)
 184            referenced.add(right)
 185
 186        orphans = node_names - referenced
 187        if orphans:
 188            fermatrica_error(
 189                f"Nodes not referenced in any edge: {sorted(orphans)}. "
 190                f"All nodes must appear in at least one edge when edges are provided."
 191            )
 192
 193        adj = {n: [] for n in node_names}
 194        for left, right, _ in self.edges:
 195            adj[left].append(right)
 196            adj[right].append(left)
 197
 198        start = next(iter(node_names))
 199        visited = set()
 200        stack = [start]
 201        while stack:
 202            current = stack.pop()
 203            if current in visited:
 204                continue
 205            visited.add(current)
 206            for neighbor in adj[current]:
 207                if neighbor not in visited:
 208                    stack.append(neighbor)
 209
 210        if len(visited) != n_nodes:
 211            unreachable = node_names - visited
 212            fermatrica_error(
 213                f"MetaModel graph is not a single connected DAG. "
 214                f"Unreachable node(s): {sorted(unreachable)}."
 215            )

 218    def _load_multi(cls, path: str) -> "MetaModel":
 219        """
 220        Load a multi-model DAG from a directory with a meta configuration.
 221
 222        Reads ``meta_conf.toml`` (the single-file format written by
 223        :meth:`MetaModel.save`), then loads each node's model + dataset +
 224        ModelRep from its corresponding subdirectory. Nodes are created in
 225        the order they appear in the graph (start nodes first, then
 226        nodes referenced in ``right``).
 227
 228        Legacy saves written as ``meta_conf.xlsx`` /
 229        ``meta_model_conf.xlsx`` (``model_graph`` sheet) remain readable —
 230        a warning is logged and the content is converted to the same
 231        structure.
 232
 233        Node folders may be named ``<name>`` or ``model_<name>`` (the
 234        layout written by :meth:`MetaModel.save`); a leading ``model_``
 235        prefix on graph names is stripped so saved nodes load with clean
 236        keys. ``model_rep.pkl.lzma`` is optional — when absent a default
 237        ``ModelRep`` is auto-created from the dataset (with a warning),
 238        matching the modelling-stage save layout. Per-node ``scope_names``
 239        are restored from the ``[node_settings.<name>]`` tables.
 240
 241        Parameters
 242        ----------
 243        path : str
 244            Directory containing the meta config file and one subdirectory
 245            per node. Each subdirectory must contain the model artifacts
 246            (as saved by ``ModelNode.save``) and, optionally,
 247            ``model_rep.pkl.lzma``.
 248
 249        Returns
 250        -------
 251        MetaModel
 252            Fully initialized with all nodes and edges.
 253
 254        Raises
 255        ------
 256        FermatricaError
 257            If the path is missing, the meta config file is not found, or
 258            a node folder referenced in the graph is missing.
 259        """
 260        if not os.path.exists(path):
 261            fermatrica_error(f"Path not found: {path}")
 262
 263        graph, settings = _read_meta_conf(path)
 264
 265        left_series = []
 266        right_series = []
 267        for entry in graph:
 268            left = entry.get("left")
 269            right = entry.get("right")
 270            if (
 271                not isinstance(left, str) or not isinstance(right, str)
 272                or not left.strip() or not right.strip()
 273            ):
 274                fermatrica_error(
 275                    "Invalid [[graph]] entry in meta config: 'left' and 'right' "
 276                    f"must be non-empty strings. Found: {entry}"
 277                )
 278            left_series.append(left.strip())
 279            right_series.append(right.strip())
 280
 281        start_nodes_set = set(left_series)
 282        finish_nodes_set = set(right_series)
 283        nodes_order = [n for n in start_nodes_set if n not in finish_nodes_set]
 284        for n in right_series:
 285            if n not in nodes_order:
 286                nodes_order.append(n)
 287
 288        name_map = {orig: _clean_node_name(orig) for orig in nodes_order}
 289
 290        nodes = {}
 291        for orig_name in nodes_order:
 292            node_key = name_map[orig_name]
 293            node_path = _resolve_node_dir(path, orig_name)
 294            if node_path is None:
 295                fermatrica_error(
 296                    f"Node folder not found for '{orig_name}' in {path}. "
 297                    "Expected '<name>' or 'model_<name>' subdirectory."
 298                )
 299
 300            nodes[node_key] = ModelNode.load(node_path)
 301
 302        for node_key, node in nodes.items():
 303            entry = settings.get(node_key)
 304            if entry is not None and "scope_names" in entry:
 305                node.scope_names = set(entry["scope_names"])
 306            if entry is not None and "static" in entry:
 307                node.static = bool(entry["static"])
 308
 309        edges = []
 310        for entry in graph:
 311            raw_left = entry["left"].strip()
 312            raw_right = entry["right"].strip()
 313            left = name_map.get(raw_left, _clean_node_name(raw_left))
 314            right = name_map.get(raw_right, _clean_node_name(raw_right))
 315            merge_val = entry.get("merge_fn")
 316            merge_fn = str(merge_val).strip() if merge_val is not None else None
 317            edges.append((left, right, merge_fn or None))
 318
 319        mm = cls(nodes=nodes, edges=edges)
 320        return mm

 322    def save(self, path: str = '', save_tag: str | None = None,
 323             include_context_path: str | None = None) -> str:
 324        """
 325        Save the MetaModel (all nodes + edges + settings) to disk.
 326
 327        Mirrors ``Model.save`` naming conventions: multi-node DAGs are
 328        written to ``path/meta_<save_tag>/`` with one ``model_<node_name>/``
 329        subdirectory per node plus a single ``meta_conf.toml`` (DAG edges
 330        as ``[[graph]]`` tables and per-node ``scope_names`` as
 331        ``[node_settings.<name>]`` tables). Single-node MetaModels
 332        delegate to :meth:`ModelNode.save` (plain
 333        ``path/model_<save_tag>/``) and write the same ``meta_conf.toml``
 334        into the node directory so that :meth:`MetaModel.load` can
 335        auto-detect the layout.
 336
 337        ``dt_pred`` is not persisted — call ``mm.compute()`` (no-arg
 338        refresh) after loading to rebuild predictions. Seasonality
 339        (``ssn_dict``) and ``media_prices`` persist automatically via the
 340        per-node ``model_rep.pkl.lzma`` pickle.
 341
 342        Adhoc module sources (``adhoc_code_src``) are snapshots taken at
 343        ``ModelRep`` construction time and pickled per node. Nodes of a
 344        MetaModel may have been saved at different development stages, so
 345        the same adhoc module can be pickled with different source across
 346        nodes — at compute time the first-imported snapshot wins
 347        process-wide. Before relying on a saved MetaModel, check that
 348        every node's saved adhoc sources are up to date with the current
 349        ``code_py/adhoc/`` files.
 350
 351        Parameters
 352        ----------
 353        path : str, default ''
 354            Path to the up-directory where the model directory is to be
 355            created. ``''`` defaults to ``code_py/model_data`` under the
 356            current working directory (same default as ``Model.save``).
 357        save_tag : str or None, optional
 358            Specific tag to name the model directory. When ``None``, a
 359            timestamp is generated exactly like ``Model.save``
 360            (``YYYY-MM-DD_HHMMSS``).
 361        include_context_path : str or None, optional
 362            Path to a context XLSX file. When provided, the file is
 363            **copied** (``shutil.copy2``) into the saved directory as
 364            ``context.xlsx`` (no sheet rewriting). Useful to make the
 365            folder self-contained for downstream reporting.
 366
 367        Returns
 368        -------
 369        str
 370            Path to the saved directory: ``meta_<save_tag>/`` for a
 371            multi-node DAG, or the single node's ``model_<save_tag>/``
 372            directory.
 373
 374        Raises
 375        ------
 376        FermatricaError
 377            If a callable merge function cannot be re-imported from its
 378            module (lambdas, closures, and ``__main__``-defined functions
 379            are not serializable).
 380        """
 381        if path == '':
 382            path = os.path.join(os.getcwd(), 'code_py', 'model_data')
 383
 384        if not os.path.isdir(path):
 385            os.makedirs(path, exist_ok=True)
 386            logging.warning(
 387                'Directory to save MetaModel is not found: ' + path +
 388                " - and created from scratch. You might delete it by hand if wasn't your intention"
 389            )
 390
 391        if save_tag is None:
 392            save_tag = re.sub(' ', '_', datetime.datetime.now().isoformat(sep=" ", timespec="seconds"))
 393            save_tag = re.sub(r':', '', save_tag)
 394        _validate_save_name(save_tag, "save_tag")
 395
 396        if len(self.nodes) == 1 and not self.edges:
 397            node_name = next(iter(self.nodes.keys()))
 398            _validate_save_name(node_name, "node name")
 399            node = next(iter(self.nodes.values()))
 400            node_dir = node.save(path=path, save_tag=save_tag)
 401            _write_meta_conf(node_dir, [], self._node_settings_map())
 402            if include_context_path:
 403                shutil.copy2(include_context_path, os.path.join(node_dir, "context.xlsx"))
 404            return node_dir
 405
 406        meta_dir = os.path.join(path, "meta_" + save_tag)
 407        if os.path.exists(meta_dir):
 408            fermatrica_error(
 409                f"Directory '{meta_dir}' already exists. "
 410                "Choose a different save_tag or remove the existing directory first."
 411            )
 412
 413        for node_name in self.nodes:
 414            _validate_save_name(node_name, "node name")
 415
 416        graph_rows = []
 417        for left, right, merge_fn in self.edges:
 418            graph_rows.append({
 419                "left": left,
 420                "right": right,
 421                "merge_fn": _serialize_merge_fn(merge_fn),
 422            })
 423
 424        os.makedirs(meta_dir, exist_ok=True)
 425
 426        try:
 427            _write_meta_conf(meta_dir, graph_rows, self._node_settings_map())
 428
 429            for node_name, node in self.nodes.items():
 430                node.save(path=meta_dir, save_tag=node_name)
 431
 432            if include_context_path:
 433                shutil.copy2(include_context_path, os.path.join(meta_dir, "context.xlsx"))
 434        except Exception:
 435            shutil.rmtree(meta_dir, ignore_errors=True)
 436            raise
 437
 438        return meta_dir

 440    def _node_settings_map(self) -> dict:
 441        """
 442        Map node name to the per-node settings dict.
 443
 444        Nodes with ``scope_names`` set and static nodes are included.
 445        Persisted via :func:`_write_meta_conf` and restored by
 446        :meth:`_load_multi` (multi-node) and ``ModelNode.load``
 447        (single-node) to restore node participation in scopes and the
 448        static (no-translation) flag.
 449        """
 450        node_settings: dict = {}
 451        for name, node in self.nodes.items():
 452            entry = {}
 453            if node.scope_names is not None:
 454                entry["scope_names"] = sorted(node.scope_names)
 455            if node.static:
 456                entry["static"] = True
 457            if entry:
 458                node_settings[name] = entry
 459        return node_settings

 486    @classmethod
 487    def load(cls, path: str) -> "MetaModel":
 488        """
 489        Load a MetaModel from a directory.
 490
 491        Auto-detects the directory type:
 492        - Contains ``meta_conf.toml`` with node subdirectories (or a
 493          legacy ``meta_conf.xlsx`` / ``meta_model_conf.xlsx``)
 494          → multi-model DAG via :meth:`_load_multi`.
 495        - ``meta_conf.toml`` without subdirectories → single model
 496          (the TOML is read by ``ModelNode.load`` to restore
 497          ``scope_names``), node name ``"model"``.
 498        - No config file, no subdirectories → single model, node name
 499          ``"model"`` via ``ModelNode.load``.
 500        - No config file but has subdirectories → raises
 501          ``FermatricaError`` (looks like a DAG missing its config).
 502
 503        Parameters
 504        ----------
 505        path : str
 506            Directory path containing the model artifacts.
 507
 508        Returns
 509        -------
 510        MetaModel
 511            Fully initialised MetaModel with all nodes and edges.
 512
 513        Raises
 514        ------
 515        FermatricaError
 516            If the path is missing, is not a directory, or the layout
 517            is ambiguous (subdirectories without meta config).
 518        """
 519        if not os.path.exists(path):
 520            fermatrica_error(f"Path not found: {path}")
 521        if not os.path.isdir(path):
 522            fermatrica_error(f"Path is not a directory: {path}")
 523
 524        has_toml = os.path.isfile(os.path.join(path, "meta_conf.toml"))
 525        legacy_names = ["meta_conf.xlsx", "meta_model_conf.xlsx"]
 526        has_legacy_conf = any(
 527            os.path.isfile(os.path.join(path, f)) for f in legacy_names
 528        )
 529
 530        subdirs = [
 531            d for d in os.listdir(path)
 532            if os.path.isdir(os.path.join(path, d))
 533            and not d.startswith("__")
 534        ]
 535
 536        if has_toml and not subdirs:
 537            node = ModelNode.load(path)
 538            return cls(nodes=node)
 539
 540        if has_toml or has_legacy_conf:
 541            return cls._load_multi(path)
 542
 543        if subdirs:
 544            fermatrica_error(
 545                f"No meta_conf.toml found in '{path}', but directory "
 546                f"contains subdirectories: {subdirs}. "
 547                f"This looks like a multi-model DAG. Add meta_conf.toml."
 548            )
 549
 550        node = ModelNode.load(path)
 551        return cls(nodes=node)

 553    def compute(
 554        self,
 555        option=None,
 556        scopes: list | None = None,
 557        allow_past: bool | None = None,
 558        if_exact: bool = False,
 559        *,
 560        return_full: bool = False,
 561    ) -> dict[str, pd.DataFrame | dict[str, dict | tuple]]:
 562        """
 563        Refresh all nodes in the DAG, or execute budget options across them.
 564
 565        When ``option`` is ``None`` (default), walks nodes in topological
 566        order, merges each predecessor's ``dt_pred`` via edge-specific merge
 567        functions, and runs the no-arg refresh (:meth:`ModelNode.compute`)
 568        on every node. Returns ``{node_name: dt_pred}``. ``scopes`` /
 569        ``return_full`` are ignored on this path. ``allow_past`` defaults to
 570        ``True`` on the refresh path (via the ``None`` sentinel) and is
 571        forwarded to the merge functions.
 572
 573        With an ``option``, uses a two-phase design to avoid redundant
 574        transform/predict work:
 575
 576        1. **Translate** — walk nodes in topological order, calling
 577           ``node.translate()`` on each. Budget values are translated into
 578           media variables cumulatively (each scope inherits prior scopes'
 579           values). No transform or predict runs — this phase only mutates
 580           ``node.ds``.
 581        2. **Compute** — walk nodes again in topological order. For each
 582           node, merge predecessor ``dt_pred`` via edge-specific merge
 583           functions (from ``meta_conf.toml``), then call
 584           ``node.compute_after_translate()`` which runs a single
 585           ``transform`` + ``predict`` and derives per-scope summaries
 586           from the shared prediction.
 587
 588        Phase separation matters for multi-model DAGs: predecessor nodes'
 589        translations complete before their predictions are merged into
 590        downstream nodes, ensuring downstream nodes see the *actual*
 591        predecessor output, not stale data.
 592
 593        ``node.ds`` and ``node.dt_pred`` are always updated on every node,
 594        regardless of ``return_full``.
 595
 596        Args:
 597            option: Option object with budget allocations. Must provide
 598                ``iter_scopes()`` yielding ``(scope_name, budget_items)``.
 599                ``None`` runs the no-arg refresh path.
 600            scopes: List of ``OptionScope`` objects defining date ranges,
 601                entity filters, reference periods, and inflation. Matched
 602                by scope name during translation. Required on the option
 603                path (scopes are not stored on the MetaModel — the same
 604                model can be computed under different scope sets).
 605            allow_past: Forwarded to ``OptionScope.build_mask`` and merge
 606                functions. When ``False`` (default on the option path), only
 607                future rows (``listed == 4``) are affected by budget
 608                translation. On the no-arg refresh path, ``None`` (default)
 609                resolves to ``True``.
 610            if_exact: If ``True``, skip long-term translation — budget
 611                values are applied only to each scope's exact period
 612                without forward projection.
 613            return_full: If ``True``, each scope entry is a 3-tuple
 614                ``(ds, dt_pred, summary)``. If ``False`` (default), each
 615                scope entry is just the ``summary`` dict. Follows the same
 616                convention as ``fermatrica.fit_predict``.
 617
 618        Returns:
 619            ``{node_name: dt_pred}`` when ``option`` is ``None``.
 620
 621            With an ``option``: ``{node_name: {scope_name: summary}}`` when
 622            ``return_full=False`` (default), or
 623            ``{node_name: {scope_name: (ds, dt_pred, summary)}}`` when
 624            ``return_full=True``.
 625
 626            Each node entry maps scope names to either a summary dict or a
 627            3-tuple. Summary dict keys include volume/value metrics such as
 628            ``pred_exact_vol``, ``pred_long_vol``, ``growth_vol``, etc.
 629
 630        Note:
 631            Predecessor merge passes the **full** predecessor ``dt_pred``
 632            (all periods), not a per-scope slice. All existing merge
 633            functions in the codebase handle this safely.
 634        """
 635        order = self._topological_order()
 636
 637        if option is None:
 638            allow_past = True if allow_past is None else allow_past
 639            results: dict[str, pd.DataFrame] = {}
 640            for node_name in order:
 641                node = self.nodes[node_name]
 642                self._merge_predecessors(node_name, allow_past)
 643                results[node_name] = node.compute()
 644            return results
 645
 646        if scopes is None:
 647            fermatrica_error(
 648                "scopes argument is required on the option path "
 649                "(pass scopes=ctx.scopes)"
 650            )
 651
 652        allow_past = False if allow_past is None else allow_past
 653
 654        for node_name in order:
 655            self.nodes[node_name].translate(option, scopes, allow_past=allow_past, if_exact=if_exact)
 656
 657        all_results: dict[str, dict] = {}
 658        for node_name in order:
 659            node = self.nodes[node_name]
 660            self._merge_predecessors(node_name, allow_past)
 661            all_results[node_name] = node.compute_after_translate(option, scopes, return_full=return_full)
 662
 663        return all_results

 665    def _merge_predecessors(self, node_name: str, allow_past: bool):
 666        """
 667        Merge each predecessor's ``dt_pred`` into the node's dataset.
 668
 669        For every direct predecessor of ``node_name`` with a registered
 670        merge function, loads the predecessor's adhoc code (if any) and
 671        applies the merge function to ``node.ds`` with the predecessor's
 672        ``dt_pred`` and the given ``allow_past``.
 673
 674        Shared by the no-arg refresh path and the option path of
 675        :meth:`compute` so both DAGs merge identical data.
 676        """
 677        node = self.nodes[node_name]
 678        for pred_name in self._predecessors(node_name):
 679            merge_fn = self._edge_merge_fn(pred_name, node_name)
 680            if merge_fn:
 681                self._load_adhoc_from_node(node)
 682                node.ds = self._apply_merge_fn(
 683                    merge_fn,
 684                    ds_cur=node.ds,
 685                    dt_pred_prev=self.nodes[pred_name].dt_pred,
 686                    allow_past=allow_past,
 687                )

 689    def _topological_order(self) -> list[str]:
 690        """
 691        Return nodes in topological order (Kahn's algorithm).
 692
 693        Start nodes (zero incoming edges) appear first, followed by
 694        nodes whose dependencies have been resolved, and finally sink
 695        nodes. Raises ``FermatricaError`` if the graph contains a
 696        cycle.
 697
 698        Returns
 699        -------
 700        list[str]
 701            Node names in execution order.
 702        """
 703        in_degree: dict[str, int] = {}
 704        adj: dict[str, list[str]] = {}
 705
 706        for node_name in self.nodes:
 707            in_degree[node_name] = 0
 708            adj[node_name] = []
 709
 710        for left, right, _ in self.edges:
 711            if left in in_degree and right in in_degree:
 712                in_degree[right] += 1
 713                adj.setdefault(left, []).append(right)
 714
 715        queue = [n for n in self.nodes if in_degree.get(n, 0) == 0]
 716        result = []
 717
 718        while queue:
 719            current = queue.pop(0)
 720            result.append(current)
 721            for neighbor in adj.get(current, []):
 722                in_degree[neighbor] -= 1
 723                if in_degree[neighbor] == 0:
 724                    queue.append(neighbor)
 725
 726        if len(result) != len(self.nodes):
 727            fermatrica_error(
 728                "MetaModel graph contains a cycle. Cannot compute."
 729            )
 730
 731        return result

 733    def _predecessors(self, node_name: str) -> list[str]:
 734        """
 735        Return the names of direct predecessors of a node.
 736
 737        Parameters
 738        ----------
 739        node_name : str
 740            Name of the node whose predecessors to look up.
 741
 742        Returns
 743        -------
 744        list[str]
 745            Names of nodes that have an edge pointing to ``node_name``.
 746            Empty list if the node has no predecessors.
 747        """
 748        return [left for left, right, _ in self.edges if right == node_name]

 750    def _edge_merge_fn(
 751        self, left: str, right: str
 752    ) -> typing.Callable | str | None:
 753        """
 754        Return the merge function for a given edge.
 755
 756        Parameters
 757        ----------
 758        left : str
 759            Name of the source (predecessor) node.
 760        right : str
 761            Name of the target (successor) node.
 762
 763        Returns
 764        -------
 765        Callable or str or None
 766            The merge function registered for the ``left -> right``
 767            edge. ``None`` if the edge has no merge function or does
 768            not exist.
 769        """
 770        for l, r, merge_fn in self.edges:
 771            if l == left and r == right:
 772                return merge_fn
 773        return None

 775    def _load_adhoc_from_node(self, node: ModelNode):
 776        """
 777        Import adhoc user modules registered on a node's ModelRep.
 778
 779        Called before invoking edge-specific merge functions so that
 780        adhoc merge functions (prefixed ``code_py.adhoc.reporting.``)
 781        are available in the current module's namespace.
 782
 783        Idempotency is keyed on the process-local module globals (see
 784        ``ModelNode._load_adhoc_code`` for the rationale): a spawn Pool
 785        worker inherits the pickled MetaModel but starts with a fresh
 786        ``fermatrica_rep.meta_model.node`` namespace, so adhoc modules
 787        must be re-imported there.
 788
 789        The imported source is the node's pickled snapshot (taken at
 790        ``ModelRep`` construction time), not the live project adhoc
 791        files. Because imports are idempotent on the module name, the
 792        first node whose snapshot registers a module wins process-wide —
 793        a DAG whose nodes were saved at different development stages can
 794        silently run stale or mixed adhoc code. Verify that every node's
 795        saved adhoc sources are up to date before relying on a loaded
 796        MetaModel.
 797
 798        Parameters
 799        ----------
 800        node : ModelNode
 801            The node whose ``model_rep.adhoc_code_src`` is imported.
 802        """
 803        if (
 804            pd.notna(node.model_rep.adhoc_code_src)
 805            and len(node.model_rep.adhoc_code_src) > 0
 806        ):
 807            for k, v in node.model_rep.adhoc_code_src.items():
 808                if k in globals():
 809                    continue
 810                fr_cur_name = inspect.currentframe().f_globals["__name__"]
 811                import_module_from_string(k, v, fr_cur_name)

 813    def _apply_merge_fn(
 814        self,
 815        merge_fn: typing.Callable | str,
 816        ds_cur: pd.DataFrame,
 817        dt_pred_prev: pd.DataFrame,
 818        allow_past: bool,
 819    ) -> pd.DataFrame:
 820        """
 821        Execute a merge function by resolving it to a callable.
 822
 823        Supports three resolution strategies:
 824        1. Already callable — invoked directly.
 825        2. String name found in ``locals()`` or ``globals()`` —
 826           resolved via ``eval()``.
 827        3. Qualified dotted name (e.g. ``module.fn``) — resolved
 828           from the module object in ``globals()``.
 829        4. (Fallback) Scans all module values for an attribute
 830           matching the name — useful for adhoc functions imported
 831           at runtime.
 832
 833        If none of the strategies resolve the function, a warning
 834        is logged and ``ds_cur`` is returned unchanged.
 835
 836        Parameters
 837        ----------
 838        merge_fn : Callable or str
 839            Merge function to invoke. Passed ``ds_cur``,
 840            ``dt_pred_prev``, and ``allow_past``.
 841        ds_cur : pd.DataFrame
 842            Current node's dataset to merge into.
 843        dt_pred_prev : pd.DataFrame
 844            Predecessor node's prediction DataFrame.
 845        allow_past : bool
 846            Whether to include historical periods in the merge.
 847
 848        Returns
 849        -------
 850        pd.DataFrame
 851            The merged dataset (modified in place or replaced).
 852        """
 853        if callable(merge_fn):
 854            return merge_fn(
 855                ds_cur=ds_cur,
 856                dt_pred_prev=dt_pred_prev,
 857                allow_past=allow_past,
 858            )
 859
 860        if merge_fn in locals() or merge_fn in globals():
 861            fn = eval(merge_fn)
 862        elif (
 863            len(merge_fn.split(".")) > 1
 864            and getattr(
 865                globals().get(
 866                    ".".join(merge_fn.split(".")[:-1]), None
 867                ),
 868                merge_fn.split(".")[-1],
 869                False,
 870            )
 871        ):
 872            fn = eval(merge_fn)
 873        else:
 874            for v in globals().values():
 875                if (
 876                    inspect.ismodule(v)
 877                    and hasattr(v, merge_fn)
 878                    and callable(getattr(v, merge_fn))
 879                ):
 880                    fn = getattr(v, merge_fn)
 881                    break
 882            else:
 883                logging.warning(
 884                    f"Merge function '{merge_fn}' not found — skipping merge"
 885                )
 886                return ds_cur
 887
 888        return fn(
 889            ds_cur=ds_cur,
 890            dt_pred_prev=dt_pred_prev,
 891            allow_past=allow_past,
 892        )

 895def _last_node_name(meta_model: MetaModel) -> str:
 896    """Return the name of the single sink (terminal) node in the DAG.
 897
 898    A sink node is a node that never appears as ``left`` in any graph
 899    edge — meaning nothing depends on it. In a chain like
 900    ``awareness → bht → sales``, ``sales`` is the sink.
 901
 902    Args:
 903        meta_model: The MetaModel whose DAG to inspect.
 904
 905    Returns:
 906        The name of the single terminal node.
 907
 908    Raises:
 909        FermatricaError: If the graph has multiple sink nodes (parallel
 910            independent branches with no common downstream consumer).
 911            Callers must pass ``summary_node`` explicitly in this case.
 912    """
 913    order = meta_model._topological_order()
 914    terminal_nodes = [
 915        n
 916        for n in order
 917        if not any(left == n for left, _, _ in meta_model.edges)
 918    ]
 919
 920    if len(terminal_nodes) > 1:
 921        fermatrica_error(
 922            f"MetaModel has {len(terminal_nodes)} sink nodes ({terminal_nodes}). "
 923            "Pass summary_node explicitly to identify the output node."
 924        )
 925
 926    return terminal_nodes[0] if terminal_nodes else order[-1]

 957def _validate_save_name(name: str, label: str) -> str:
 958    """Validate a ``save_tag`` / node name used as a directory component.
 959
 960    Names must be non-empty, contain no path separators, no ``..``, and no
 961    characters invalid in directory names — otherwise the saved tree could
 962    escape ``path`` or be un-loadable. Raises ``FermatricaError``.
 963    """
 964    if not name or not str(name).strip():
 965        fermatrica_error(f"{label} must be a non-empty string")
 966    if isinstance(name, str) and name != name.strip():
 967        fermatrica_error(f"{label} '{name}' must not have leading/trailing whitespace")
 968    invalid_chars = {os.sep, "/"}
 969    if os.altsep:
 970        invalid_chars.add(os.altsep)
 971    invalid_chars.update('< > : " | ? * \\'.split())
 972    if any(c in str(name) for c in invalid_chars) or str(name) in (".", ".."):
 973        fermatrica_error(
 974            f"{label} '{name}' contains invalid characters — it is used as a directory name"
 975        )
 976    return name

 979def _serialize_merge_fn(merge_fn: typing.Callable | str | None) -> str | None:
 980    """Serialize a merge function into its ``meta_conf.toml`` value.
 981
 982    - ``None`` → ``None`` (omitted → parallel branch).
 983    - ``str`` → written as-is.
 984    - callable → qualified name ``f"{module}.{qualname}"``, verified to be
 985      re-importable. Lambdas, closures, and ``__main__``-defined functions
 986      raise ``FermatricaError`` (dropping a merge silently is worse).
 987    """
 988    if merge_fn is None or isinstance(merge_fn, str):
 989        return merge_fn
 990
 991    qualname = f"{merge_fn.__module__}.{merge_fn.__qualname__}"
 992    module_name = merge_fn.__module__
 993    parts = merge_fn.__qualname__.split(".")
 994    try:
 995        obj = importlib.import_module(module_name)
 996        for part in parts:
 997            obj = getattr(obj, part)
 998        if not callable(obj):
 999            raise AttributeError
1000    except (ImportError, AttributeError) as exc:
1001        fermatrica_error(
1002            f"Merge function '{qualname}' cannot be re-imported from its module. "
1003            "MetaModel.save requires merge functions to be top-level, importable "
1004            f"callables. Original error: {exc}"
1005        )
1006    return qualname

1009def _toml_basic_string(value: str) -> str:
1010    """Encode ``value`` as a TOML basic string."""
1011    out = []
1012    for ch in value:
1013        code = ord(ch)
1014        if ch == '"':
1015            out.append('\\"')
1016        elif ch == "\\":
1017            out.append("\\\\")
1018        elif ch == "\n":
1019            out.append("\\n")
1020        elif ch == "\t":
1021            out.append("\\t")
1022        elif ch == "\r":
1023            out.append("\\r")
1024        elif ch == "\f":
1025            out.append("\\f")
1026        elif ch == "\b":
1027            out.append("\\b")
1028        elif code < 0x20 or code == 0x7F:
1029            out.append(f"\\u{code:04X}")
1030        else:
1031            out.append(ch)
1032    return '"' + "".join(out) + '"'

1035def _write_meta_conf(meta_dir: str, graph_rows: list[dict], node_settings: dict) -> str:
1036    """
1037    Write the DAG graph + per-node settings into a single ``meta_conf.toml``.
1038
1039    Each edge becomes a ``[[graph]]`` table (``left``, ``right`` and an
1040    optional ``merge_fn``); each node with settings becomes a
1041    ``[node_settings.<name>]`` table carrying ``scope_names`` (when set)
1042    and ``static`` (when true). Returns the path of the written file.
1043    """
1044    lines: list[str] = []
1045    for row in graph_rows:
1046        lines.append("[[graph]]")
1047        lines.append(f"left = {_toml_basic_string(row['left'])}")
1048        lines.append(f"right = {_toml_basic_string(row['right'])}")
1049        if row.get("merge_fn"):
1050            lines.append(f"merge_fn = {_toml_basic_string(row['merge_fn'])}")
1051        lines.append("")
1052    for name, entry in node_settings.items():
1053        lines.append(f"[node_settings.{_toml_basic_string(name)}]")
1054        if entry.get("scope_names") is not None:
1055            names = sorted(entry["scope_names"])
1056            lines.append("scope_names = [" + ", ".join(_toml_basic_string(n) for n in names) + "]")
1057        if entry.get("static"):
1058            lines.append("static = true")
1059        lines.append("")
1060
1061    path = os.path.join(meta_dir, "meta_conf.toml")
1062    with open(path, "w", encoding="utf-8") as handle:
1063        handle.write("\n".join(lines))
1064    return path

1067def _read_meta_conf(path: str) -> tuple[list[dict], dict]:
1068    """
1069    Read the DAG graph + per-node settings from a meta config file.
1070
1071    Prefers the single-file ``meta_conf.toml``; falls back to the legacy
1072    ``meta_conf.xlsx`` / ``meta_model_conf.xlsx`` (``model_graph`` sheet),
1073    with a warning, for directories saved before the TOML format. Raises
1074    ``FermatricaError`` when neither is present.
1075
1076    Returns a ``(graph, settings)`` pair: ``graph`` is a list of
1077    ``{"left", "right", "merge_fn"?}`` dicts, ``settings`` maps node name
1078    to a settings dict.
1079    """
1080    toml_path = os.path.join(path, "meta_conf.toml")
1081    if os.path.exists(toml_path):
1082        with open(toml_path, "rb") as handle:
1083            conf = tomllib.load(handle)
1084        return conf.get("graph", []), conf.get("node_settings", {})
1085
1086    legacy_names = ["meta_conf.xlsx", "meta_model_conf.xlsx"]
1087    found = [f for f in legacy_names if os.path.isfile(os.path.join(path, f))]
1088    if not found:
1089        fermatrica_error(
1090            f"No meta_conf.toml found in {path} "
1091            "(and no legacy meta_conf.xlsx / meta_model_conf.xlsx either)"
1092        )

1095        f"Reading legacy meta config from {found[0]} — re-save via MetaModel.save "

1097    )

1099        os.path.join(path, found[0]), sheet_name="model_graph", engine="openpyxl", dtype=str
1100    )
1101
1102    if "merge_fn" not in model_graph.columns:
1103        logging.warning(
1104            "column 'merge_fn' not found in model_graph sheet — assuming no merge functions"
1105        )

1112        merge_val = row.get("merge_fn")
1113        if pd.notna(merge_val) and str(merge_val).strip():
1114            entry["merge_fn"] = str(merge_val).strip()

1150def _convert_to_metamodel(model, ds=None, model_rep=None, dt_pred=None, merge_fn_pattern=None):
1151    """
1152    Convert ``Model | ModelNode | MetaModel | ModelRep | list`` + optional
1153    overrides into a ``MetaModel``.
1154
1155    Always produces a ``MetaModel``; the caller can iterate
1156    ``mm.nodes.items()`` and pass ``(node.model, node.ds,
1157    node.model_rep, node.dt_pred)`` to their worker.
1158
1159    Parameters
1160    ----------
1161    merge_fn_pattern : str | None
1162        When ``model`` is a ``list`` and this is a string like
1163        ``"code_py.adhoc.reporting.pred_merge_{i}"``, sequential edges
1164        are created between consecutive list elements with merge function
1165        names interpolated from the pattern.  When ``None`` (default),
1166        no edges are created — nodes are independent.
1167
1168    When ``model_rep`` is ``None``, a default ``ModelRep`` is auto-created
1169    from ``ds`` (falling back to ``dt_pred``, then an empty frame) via
1170    ``_auto_model_rep``.
1171    """
1172    if isinstance(model, MetaModel):
1173        return model
1174
1175    if isinstance(model, list):
1176        warnings.warn(
1177            "Passing list of models is deprecated. Use MetaModel instead.",
1178            DeprecationWarning,
1179            stacklevel=3,
1180        )
1181        nodes = {}
1182        for i, m in enumerate(model):
1183            if isinstance(m, ModelRep):
1184                nodes[f"model_{i}"] = ModelNode(
1185                    model=None, ds=pd.DataFrame(), model_rep=m, dt_pred=None,
1186                    _display_only=True,
1187                )
1188            else:
1189                _ds = ds[i] if isinstance(ds, (list, tuple)) else ds
1190                if _ds is None:
1191                    _ds = pd.DataFrame()
1192                _rep = model_rep[i] if isinstance(model_rep, (list, tuple)) else model_rep
1193                _pred = dt_pred[i] if isinstance(dt_pred, (list, tuple)) else dt_pred
1194                if _rep is None:
1195                    _rep = _auto_model_rep(_ds, _pred)
1196                nodes[f"model_{i}"] = ModelNode(model=m, ds=_ds, model_rep=_rep, dt_pred=_pred)
1197
1198        edges = []
1199        if merge_fn_pattern:
1200            for i in range(len(model) - 1):
1201                edges.append((f"model_{i}", f"model_{i + 1}", merge_fn_pattern.format(i=i + 1)))
1202
1203        return MetaModel(nodes=nodes, edges=edges)
1204
1205    if isinstance(model, ModelNode):
1206        node = model
1207        if ds is not None or model_rep is not None or dt_pred is not None:
1208            node = ModelNode(
1209                model=node.model,
1210                ds=ds if ds is not None else node.ds,

142    def _validate(self):
143        """
144        Validate node and edge consistency.
145
146        Raises FermatricaError if the graph is malformed.
147        """
148        node_names = set(self.nodes.keys())
149        n_nodes = len(node_names)
150
151        if n_nodes == 0:
152            fermatrica_error("MetaModel must contain at least one node.")
153
154        if n_nodes > 1 and not self.edges:
155            fermatrica_error(
156                f"MetaModel has {n_nodes} nodes but no edges provided. "
157                "Edges are required when more than one node is present."
158            )
159
160        if not self.edges:
161            return
162
163        for left, right, _ in self.edges:
164            if left not in node_names:
165                fermatrica_error(
166                    f"Edge left node '{left}' is not in MetaModel nodes: {sorted(node_names)}"
167                )
168            if right not in node_names:
169                fermatrica_error(
170                    f"Edge right node '{right}' is not in MetaModel nodes: {sorted(node_names)}"
171                )
172            if left == right:
173                fermatrica_error(f"Self-loop detected: '{left}' -> '{right}'.")
174
175        seen_edges = set()
176        for edge in self.edges:
177            key = (edge[0], edge[1])
178            if key in seen_edges:
179                fermatrica_error(f"Duplicate edge: '{key[0]}' -> '{key[1]}'.")
180            seen_edges.add(key)
181
182        referenced = set()
183        for left, right, _ in self.edges:
184            referenced.add(left)
185            referenced.add(right)
186
187        orphans = node_names - referenced
188        if orphans:
189            fermatrica_error(
190                f"Nodes not referenced in any edge: {sorted(orphans)}. "
191                f"All nodes must appear in at least one edge when edges are provided."
192            )
193
194        adj = {n: [] for n in node_names}
195        for left, right, _ in self.edges:
196            adj[left].append(right)
197            adj[right].append(left)
198
199        start = next(iter(node_names))
200        visited = set()
201        stack = [start]
202        while stack:
203            current = stack.pop()
204            if current in visited:
205                continue
206            visited.add(current)
207            for neighbor in adj[current]:
208                if neighbor not in visited:
209                    stack.append(neighbor)
210
211        if len(visited) != n_nodes:
212            unreachable = node_names - visited
213            fermatrica_error(
214                f"MetaModel graph is not a single connected DAG. "
215                f"Unreachable node(s): {sorted(unreachable)}."
216            )

241
242        Parameters
243        ----------
244        path : str
245            Directory containing the meta config file and one subdirectory
246            per node. Each subdirectory must contain the model artifacts
247            (as saved by ``ModelNode.save``) and, optionally,
248            ``model_rep.pkl.lzma``.
249
250        Returns
251        -------
252        MetaModel
253            Fully initialized with all nodes and edges.
254
255        Raises
256        ------
257        FermatricaError

259            a node folder referenced in the graph is missing.
260        """
261        if not os.path.exists(path):
262            fermatrica_error(f"Path not found: {path}")
263

267        right_series = []

270            right = entry.get("right")

272                not isinstance(left, str) or not isinstance(right, str)
273                or not left.strip() or not right.strip()

276                    "Invalid [[graph]] entry in meta config: 'left' and 'right' "
277                    f"must be non-empty strings. Found: {entry}"
278                )
279            left_series.append(left.strip())
280            right_series.append(right.strip())
281
282        start_nodes_set = set(left_series)
283        finish_nodes_set = set(right_series)
284        nodes_order = [n for n in start_nodes_set if n not in finish_nodes_set]
285        for n in right_series:
286            if n not in nodes_order:

288

307            if entry is not None and "static" in entry:

353        ----------

379            module (lambdas, closures, and ``__main__``-defined functions

383            path = os.path.join(os.getcwd(), 'code_py', 'model_data')

396
397        if len(self.nodes) == 1 and not self.edges:

399            _validate_save_name(node_name, "node name")

410                f"Directory '{meta_dir}' already exists. "

416

489        """
490        Load a MetaModel from a directory.
491
492        Auto-detects the directory type:
493        - Contains ``meta_conf.toml`` with node subdirectories (or a
494          legacy ``meta_conf.xlsx`` / ``meta_model_conf.xlsx``)
495          → multi-model DAG via :meth:`_load_multi`.
496        - ``meta_conf.toml`` without subdirectories → single model
497          (the TOML is read by ``ModelNode.load`` to restore
498          ``scope_names``), node name ``"model"``.
499        - No config file, no subdirectories → single model, node name
500          ``"model"`` via ``ModelNode.load``.
501        - No config file but has subdirectories → raises
502          ``FermatricaError`` (looks like a DAG missing its config).
503
504        Parameters
505        ----------
506        path : str
507            Directory path containing the model artifacts.
508
509        Returns
510        -------
511        MetaModel
512            Fully initialised MetaModel with all nodes and edges.
513
514        Raises
515        ------
516        FermatricaError
517            If the path is missing, is not a directory, or the layout
518            is ambiguous (subdirectories without meta config).
519        """
520        if not os.path.exists(path):
521            fermatrica_error(f"Path not found: {path}")
522        if not os.path.isdir(path):
523            fermatrica_error(f"Path is not a directory: {path}")
524
525        has_toml = os.path.isfile(os.path.join(path, "meta_conf.toml"))
526        legacy_names = ["meta_conf.xlsx", "meta_model_conf.xlsx"]
527        has_legacy_conf = any(
528            os.path.isfile(os.path.join(path, f)) for f in legacy_names
529        )
530
531        subdirs = [
532            d for d in os.listdir(path)
533            if os.path.isdir(os.path.join(path, d))
534            and not d.startswith("__")
535        ]
536
537        if has_toml and not subdirs:
538            node = ModelNode.load(path)
539            return cls(nodes=node)
540
541        if has_toml or has_legacy_conf:
542            return cls._load_multi(path)
543
544        if subdirs:
545            fermatrica_error(
546                f"No meta_conf.toml found in '{path}', but directory "
547                f"contains subdirectories: {subdirs}. "
548                f"This looks like a multi-model DAG. Add meta_conf.toml."
549            )
550
551        node = ModelNode.load(path)
552        return cls(nodes=node)

554    def compute(
555        self,
556        option=None,
557        scopes: list | None = None,
558        allow_past: bool | None = None,
559        if_exact: bool = False,
560        *,
561        return_full: bool = False,
562    ) -> dict[str, pd.DataFrame | dict[str, dict | tuple]]:
563        """
564        Refresh all nodes in the DAG, or execute budget options across them.
565
566        When ``option`` is ``None`` (default), walks nodes in topological
567        order, merges each predecessor's ``dt_pred`` via edge-specific merge
568        functions, and runs the no-arg refresh (:meth:`ModelNode.compute`)
569        on every node. Returns ``{node_name: dt_pred}``. ``scopes`` /
570        ``return_full`` are ignored on this path. ``allow_past`` defaults to
571        ``True`` on the refresh path (via the ``None`` sentinel) and is
572        forwarded to the merge functions.
573
574        With an ``option``, uses a two-phase design to avoid redundant
575        transform/predict work:
576
577        1. **Translate** — walk nodes in topological order, calling
578           ``node.translate()`` on each. Budget values are translated into
579           media variables cumulatively (each scope inherits prior scopes'
580           values). No transform or predict runs — this phase only mutates
581           ``node.ds``.
582        2. **Compute** — walk nodes again in topological order. For each
583           node, merge predecessor ``dt_pred`` via edge-specific merge
584           functions (from ``meta_conf.toml``), then call
585           ``node.compute_after_translate()`` which runs a single
586           ``transform`` + ``predict`` and derives per-scope summaries
587           from the shared prediction.
588
589        Phase separation matters for multi-model DAGs: predecessor nodes'
590        translations complete before their predictions are merged into
591        downstream nodes, ensuring downstream nodes see the *actual*
592        predecessor output, not stale data.
593
594        ``node.ds`` and ``node.dt_pred`` are always updated on every node,
595        regardless of ``return_full``.
596
597        Args:
598            option: Option object with budget allocations. Must provide
599                ``iter_scopes()`` yielding ``(scope_name, budget_items)``.
600                ``None`` runs the no-arg refresh path.
601            scopes: List of ``OptionScope`` objects defining date ranges,
602                entity filters, reference periods, and inflation. Matched
603                by scope name during translation. Required on the option
604                path (scopes are not stored on the MetaModel — the same
605                model can be computed under different scope sets).
606            allow_past: Forwarded to ``OptionScope.build_mask`` and merge
607                functions. When ``False`` (default on the option path), only
608                future rows (``listed == 4``) are affected by budget
609                translation. On the no-arg refresh path, ``None`` (default)
610                resolves to ``True``.
611            if_exact: If ``True``, skip long-term translation — budget
612                values are applied only to each scope's exact period
613                without forward projection.
614            return_full: If ``True``, each scope entry is a 3-tuple
615                ``(ds, dt_pred, summary)``. If ``False`` (default), each
616                scope entry is just the ``summary`` dict. Follows the same
617                convention as ``fermatrica.fit_predict``.
618
619        Returns:
620            ``{node_name: dt_pred}`` when ``option`` is ``None``.
621
622            With an ``option``: ``{node_name: {scope_name: summary}}`` when
623            ``return_full=False`` (default), or
624            ``{node_name: {scope_name: (ds, dt_pred, summary)}}`` when
625            ``return_full=True``.
626
627            Each node entry maps scope names to either a summary dict or a

666    def _merge_predecessors(self, node_name: str, allow_past: bool):
667        """
668        Merge each predecessor's ``dt_pred`` into the node's dataset.
669
670        For every direct predecessor of ``node_name`` with a registered
671        merge function, loads the predecessor's adhoc code (if any) and
672        applies the merge function to ``node.ds`` with the predecessor's
673        ``dt_pred`` and the given ``allow_past``.
674
675        Shared by the no-arg refresh path and the option path of
676        :meth:`compute` so both DAGs merge identical data.
677        """
678        node = self.nodes[node_name]
679        for pred_name in self._predecessors(node_name):
680            merge_fn = self._edge_merge_fn(pred_name, node_name)
681            if merge_fn:
682                self._load_adhoc_from_node(node)
683                node.ds = self._apply_merge_fn(
684                    merge_fn,
685                    ds_cur=node.ds,

687                    allow_past=allow_past,

690    def _topological_order(self) -> list[str]:
691        """
692        Return nodes in topological order (Kahn's algorithm).
693
694        Start nodes (zero incoming edges) appear first, followed by
695        nodes whose dependencies have been resolved, and finally sink
696        nodes. Raises ``FermatricaError`` if the graph contains a
697        cycle.
698
699        Returns
700        -------
701        list[str]
702            Node names in execution order.
703        """
704        in_degree: dict[str, int] = {}
705        adj: dict[str, list[str]] = {}
706
707        for node_name in self.nodes:
708            in_degree[node_name] = 0
709            adj[node_name] = []
710
711        for left, right, _ in self.edges:
712            if left in in_degree and right in in_degree:
713                in_degree[right] += 1
714                adj.setdefault(left, []).append(right)
715
716        queue = [n for n in self.nodes if in_degree.get(n, 0) == 0]
717        result = []
718
719        while queue:
720            current = queue.pop(0)
721            result.append(current)
722            for neighbor in adj.get(current, []):
723                in_degree[neighbor] -= 1
724                if in_degree[neighbor] == 0:
725                    queue.append(neighbor)
726
727        if len(result) != len(self.nodes):
728            fermatrica_error(
729                "MetaModel graph contains a cycle. Cannot compute."
730            )
731
732        return result

814    def _apply_merge_fn(
815        self,
816        merge_fn: typing.Callable | str,
817        ds_cur: pd.DataFrame,
818        dt_pred_prev: pd.DataFrame,
819        allow_past: bool,
820    ) -> pd.DataFrame:
821        """
822        Execute a merge function by resolving it to a callable.
823
824        Supports three resolution strategies:
825        1. Already callable — invoked directly.
826        2. String name found in ``locals()`` or ``globals()`` —
827           resolved via ``eval()``.
828        3. Qualified dotted name (e.g. ``module.fn``) — resolved
829           from the module object in ``globals()``.
830        4. (Fallback) Scans all module values for an attribute
831           matching the name — useful for adhoc functions imported
832           at runtime.
833
834        If none of the strategies resolve the function, a warning
835        is logged and ``ds_cur`` is returned unchanged.
836
837        Parameters
838        ----------
839        merge_fn : Callable or str
840            Merge function to invoke. Passed ``ds_cur``,
841            ``dt_pred_prev``, and ``allow_past``.
842        ds_cur : pd.DataFrame
843            Current node's dataset to merge into.
844        dt_pred_prev : pd.DataFrame
845            Predecessor node's prediction DataFrame.
846        allow_past : bool
847            Whether to include historical periods in the merge.
848
849        Returns
850        -------
851        pd.DataFrame
852            The merged dataset (modified in place or replaced).
853        """
854        if callable(merge_fn):
855            return merge_fn(
856                ds_cur=ds_cur,
857                dt_pred_prev=dt_pred_prev,
858                allow_past=allow_past,
859            )
860
861        if merge_fn in locals() or merge_fn in globals():
862            fn = eval(merge_fn)
863        elif (
864            len(merge_fn.split(".")) > 1
865            and getattr(
866                globals().get(
867                    ".".join(merge_fn.split(".")[:-1]), None
868                ),
869                merge_fn.split(".")[-1],
870                False,
871            )
872        ):
873            fn = eval(merge_fn)
874        else:
875            for v in globals().values():
876                if (
877                    inspect.ismodule(v)
878                    and hasattr(v, merge_fn)
879                    and callable(getattr(v, merge_fn))
880                ):
881                    fn = getattr(v, merge_fn)
882                    break
883            else:
884                logging.warning(
885                    f"Merge function '{merge_fn}' not found — skipping merge"
886                )
887                return ds_cur
888
889        return fn(
890            ds_cur=ds_cur,
891            dt_pred_prev=dt_pred_prev,
892            allow_past=allow_past,
893        )

Load a MetaModel from a directory.

Auto-detects the directory type:

  • Contains meta_conf.toml with node subdirectories (or a legacy meta_conf.xlsx / meta_model_conf.xlsx) → multi-model DAG via _load_multi().
  • meta_conf.toml without subdirectories → single model (the TOML is read by ModelNode.load to restore scope_names), node name "model".
  • No config file, no subdirectories → single model, node name "model" via ModelNode.load.
  • No config file but has subdirectories → raises FermatricaError (looks like a DAG missing its config).

Parameters

path : str Directory path containing the model artifacts.

Returns

MetaModel Fully initialised MetaModel with all nodes and edges.

Raises

FermatricaError If the path is missing, is not a directory, or the layout is ambiguous (subdirectories without meta config).
