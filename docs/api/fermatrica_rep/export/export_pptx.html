  79def options_table(prs,
  80                  meta_model: MetaModel,
  81                  scopes: list,
  82                   options: dict | list,
  83                  *,
  84                  brands: list | None = None,
  85                  brands_var: str = 'superbrand',
  86                  filters: dict | None = None,
  87                  if_exact: bool = True,
  88                  summary_node: str | None = None,
  89                  cores: int = 1,
  90                  ):
  91    """
  92    Export options summary as a table (new API only).
  93
  94    Per scope, when the zero option has no budget/result bound to it (p09
  95    per-option scopes), the slide is still generated with the option's absolute
  96    KPIs but the zero-dependent comparison cells (incremental sales, profit,
  97    ROMI, long-term growth) are rendered as ``'-'``; scopes bound only to the
  98    zero option are reported the same way (zero baseline KPIs, unbound options
  99    as ``'-'``).
 100
 101    Filtering (user-filters-replace-default semantics): when `brands` and
 102    `filters` are both None, the filter defaults to the target brand's bs_keys
 103    ``{'bs_key': ...}`` (current behavior); passing any user filter parameter
 104    uses only the user's filters.
 105
 106    :param prs: Presentation object from python_pptx package
 107    :param meta_model: MetaModel DAG
 108    :param scopes: list of OptionScope objects
 109    :param options: dict of option name to budget data ({scope_name: {item: value}}), or list of Option objects
 110    :param brands: simple filter values, applied to `brands_var`
 111    :param brands_var: column the simple filter values apply to
 112    :param filters: complex filter ``{column: [values]}``
 113    :param if_exact: apply only to specific time period, without next years
 114    :param summary_node: name of the DAG node providing KPI metrics
 115    :param cores: number of processor cores for compute_multi (default 1 for safety)
 116    :return: Presentation object from python_pptx package
 117    """
 118    summary_node = summary_node or _last_node_name(meta_model)
 119
 120    if brands is None and filters is None:
 121        filters = _auto_bs_key_filters(meta_model, summary_node)
 122
 123    prs = slides.options.create(prs=prs,
 124                         meta_model=meta_model,
 125                         scopes=scopes,
 126                         options=options,
 127                         if_exact=if_exact,
 128                         brands=brands,
 129                         brands_var=brands_var,
 130                         filters=filters,
 131                         summary_node=summary_node,
 132                         cores=cores)
 133
 134    return prs

 330def _scope_group_meta(scopes_eff):
 331    """date_start/date_end (union), group_var, target_vals (common filter values)."""
 332    date_start = min(s.date_start for s in scopes_eff)
 333    date_end = max(s.date_end for s in scopes_eff)
 334    keys_sets = [set(s.filters.keys()) for s in scopes_eff]
 335    common_keys = set.intersection(*keys_sets) if keys_sets else set()
 336    const_keys = [k for k in scopes_eff[0].filters
 337                  if k in common_keys
 338                  and len({tuple(sorted(s.filters[k])) for s in scopes_eff}) == 1]
 339    if const_keys:
 340        group_var = const_keys
 341        target_vals = [v for k in const_keys
 342                       for v in sorted(set().union(*(s.filters[k] for s in scopes_eff)))]
 343    else:
 344        group_var = ['superbrand']
 345        target_vals = []
 346    return {'date_start': date_start, 'date_end': date_end,
 347            'group_var': group_var, 'target_vals': target_vals}

 350def _sum_budgets(option, scopes_eff):
 351    """Sum budget items across scopes. Non-Option (dict) passed through unchanged."""
 352    if not hasattr(option, 'budget_for_scope'):
 353        return option
 354    items = set()
 355    for s in scopes_eff:
 356        items |= set(option.budget_for_scope(s.scope_name).keys())
 357    return {item: sum(option.budget_for_scope(s.scope_name).get(item, 0.0)
 358                      for s in scopes_eff) for item in items}

 361def _union_scope_filters(scopes_eff):
 362    """{col: sorted unique union of values} across the scope set."""
 363    union = {}
 364    for s in scopes_eff:
 365        for k, vals in s.filters.items():
 366            union.setdefault(k, set()).update(vals)
 367    return {k: sorted(v) for k, v in union.items()}

 370def _merge_filter_dicts(base: dict | None, extra: dict | None) -> dict:
 371    """Merge two ``{col: [values]}`` filter dicts; same-key values are
 372    intersected (AND semantics)."""
 373    merged = dict(base) if base else {}
 374    for col, values in (extra or {}).items():
 375        if col in merged:
 376            merged[col] = sorted(set(merged[col]) & set(values))
 377        else:
 378            merged[col] = list(values)
 379    return merged

 382def _find_scope_for_row(scopes_eff, row, sku_var):
 383    """Scope matching the most filter keys whose values equal row's sku_var values."""
 384    best, best_score = None, -1
 385    for s in scopes_eff:
 386        score = sum(1 for k, vals in s.filters.items()
 387                    if k in sku_var and row.get(k) in vals)
 388        if score > best_score:
 389            best, best_score = s, score
 390    return best

 393def _sku_levels_expand(sku_var, group_var, if_sku_hierarchy):
 394    """Expand sku_var into the prefix levels to report.
 395
 396    :param sku_var: columns to report as "SKU" (coarse -> fine order)
 397    :param group_var: brand-level group columns already reported on a top-level slide
 398    :param if_sku_hierarchy: report every prefix level of sku_var (True) or only the
 399        bottom (full) combination (False, legacy behavior)
 400    :return: list[list[str]] of column levels, in coarse -> fine order
 401    """
 402    if not sku_var:
 403        return []
 404    if not if_sku_hierarchy:
 405        return [list(sku_var)]
 406    group_set = set(group_var)
 407    levels = [list(sku_var[:k]) for k in range(1, len(sku_var) + 1)]
 408    levels = [level for level in levels if not set(level) <= group_set]
 409    if not levels:
 410        levels = [list(sku_var)]
 411    return levels

 414def _iter_sku_level_keys(dt_pred_cur, superbrand, level):
 415    """Yield (key_row, bs_keys) for every deduped combination of `level` columns
 416    belonging to `superbrand` in dt_pred_cur."""
 417    sub = dt_pred_cur[dt_pred_cur['superbrand'] == superbrand]
 418    df = sub[level].drop_duplicates()
 419    for _, key_row in df.iterrows():
 420        bs_keys = key_row.to_frame().transpose().merge(sub, on=level, how='left')['bs_key'].unique()
 421        yield key_row, bs_keys

 424def _generate_node_slides(prs, node, ds_cur, dt_pred_cur,
 425                          option, opt_title, if_retro, node_scopes,
 426                          sku_var, period, if_volume, if_fit,
 427                          if_decompose, if_waterfall,
 428                          if_sku_fit_predict, if_sku_decompose,
 429                          if_sku_waterfall, custom_extract_effect,
 430                          show_pretrain, if_debug, listed_filter,
 431                          group_var, if_sku_hierarchy=False,
 432                          filters=None,
 433                          meta_model=None,
 434                          node_name=None,
 435                          whole_graph=False):
 436
 437    superbrand = node.model.conf.target_superbrand
 438
 439    ds_cur = ds_cur[ds_cur['listed'].isin(listed_filter)]
 440    dt_pred_cur = dt_pred_cur[dt_pred_cur['listed'].isin(listed_filter)]
 441    ds_cur = pandas_force_materialize(ds_cur, if_hard=False, if_soft=True)
 442
 443    if sku_var:
 444        existing_sku_vars = [v for v in sku_var if v in dt_pred_cur.columns]
 445        if len(existing_sku_vars) != len(sku_var):
 446            import logging
 447            logging.warning(f"Skipping SKU slides for node: dt_pred_cur missing columns {set(sku_var) - set(existing_sku_vars)}")
 448            existing_sku_vars = None
 449    else:
 450        existing_sku_vars = None
 451
 452    if custom_extract_effect is None:
 453        if whole_graph and node_name:
 454            split_m_m = extract_effect(meta_model, whole_graph=node_name,
 455                                       listed_filter=listed_filter, if_debug=if_debug)
 456        else:
 457            split_m_m = extract_effect(node, ds=ds_cur,
 458                                       listed_filter=listed_filter, if_debug=if_debug)
 459    else:
 460        split_m_m = custom_extract_effect(node, ds=ds_cur)
 461
 462    if if_fit:
 463        option_cur = _sum_budgets(option, node_scopes) if len(node_scopes) > 1 else option
 464        prs = slides.fit_predict.create(prs=prs,
 465                                        node=node,
 466                                        dt_pred=dt_pred_cur,
 467                                        option_name=opt_title,
 468                                        opt_set=node_scopes[0],
 469                                        option=option_cur,
 470                                        period=period,
 471                                        show_future=True,
 472                                        show_pretrain=show_pretrain,
 473                                        group_var=group_var,
 474                                        plot_type='brand' + '_retro' * if_retro,
 475                                        filters=filters)
 476
 477    if if_decompose:
 478        prs = slides.decomposition.create(prs=prs,
 479                                          node=node,
 480                                          split_m_m=split_m_m,
 481                                          option_name=opt_title,
 482                                          brands=[superbrand],
 483                                          period=period,
 484                                          show_future=True,
 485                                          show_pretrain=show_pretrain,
 486                                          group_var=group_var,
 487                                          plot_type='brand' + '_retro' * if_retro,
 488                                          if_volume=if_volume,
 489                                          filters=filters)
 490
 491    if if_waterfall:
 492        date_start = str(min(s.date_start for s in node_scopes).date())
 493        date_end = str(max(s.date_end for s in node_scopes).date())
 494
 495        prs = slides.waterfall.create(prs=prs,
 496                                      node=node,
 497                                      split_m_m=split_m_m,
 498                                      date_start=date_start,
 499                                      date_end=date_end,
 500                                      option_name=opt_title,
 501                                      brands=[superbrand],
 502                                      absolute_sort=False,
 503                                      plot_type='brand' + '_retro' * if_retro,
 504                                      if_volume=if_volume,
 505                                      show_pretrain=show_pretrain,
 506                                      filters=_merge_filter_dicts(filters,
 507                                                                  _union_scope_filters(node_scopes)))
 508
 509    if existing_sku_vars and any([if_sku_fit_predict, if_sku_decompose, if_sku_waterfall]):
 510        levels = _sku_levels_expand(existing_sku_vars, group_var, if_sku_hierarchy)
 511        level_keys = {tuple(level): list(_iter_sku_level_keys(dt_pred_cur, superbrand, level))
 512                      for level in levels}
 513        n_levels = sum(len(items) for items in level_keys.values())
 514        if n_levels > 50:
 515            warnings.warn(
 516                f"_generate_node_slides: {n_levels} SKU slide sets across "
 517                f"{len(levels)} hierarchy levels exceed 50. Consider narrowing sku_var "
 518                "or disabling if_sku_hierarchy.",
 519                stacklevel=2,
 520            )
 521
 522        for level in levels:
 523            for key_row, bs_keys in level_keys[tuple(level)]:
 524                try:
 525                    if option is not None:

1044def _set_config_fonts(model_rep: "ModelRep"
1045                      , theme_xml
1046                      , a_nmsp: dict):
1047    """
1048    Extract fonts from presentation template XML and set to ModelRep reporting settings.
1049
1050    :param model_rep: ModelRep (reporting settings)
1051    :param theme_xml: ElementTree XML object form `libxml` package containing theme info
1052    :param a_nmsp: namespace dictionary (required by xpath)
1053    :return: ModelRep reporting settings object
1054    """
1055
1056    model_rep.pptx_cnf["font_family_header"] = theme_xml.xpath('//a:majorFont/a:latin/@typeface', namespaces=a_nmsp)[0]
1057    model_rep.pptx_cnf["font_family_body"] = theme_xml.xpath('//a:minorFont/a:latin/@typeface', namespaces=a_nmsp)[0]
1058
1059    # Font sizes are stored as plain int (EMU) rather than Pt objects.
1060    # Pt.__reduce_ex__ from python-pptx is broken: it serialises the
1061    # internal EMU value as the constructor argument, causing deepcopy
1062    # and pickle to multiply by 12700 a second time on reconstruction.
1063    # Computing in point-space and multiplying by the EMU factor directly
1064    # avoids Pt and preserves correct values through copy / pickle.
1065    _pt_main = round(model_rep.pptx_cnf["slide_height"] / 5.5e+5)
1066    model_rep.pptx_cnf["font_size_main"] = _pt_main * 12700
1067    model_rep.pptx_cnf["font_size_small"] = round(_pt_main / 1.2) * 12700
