fermatrica_rep.export.export_summary

Brand summary XLSX export.

For every brand in a MetaModel DAG, exports one XLSX file containing:

  • basic model info (model type, prices, distribution / conf tables, per-brand stats)
  • full efficiency curves (short- and long-term: incremental volume/value, profit, ROI)
  • dynamic decomposition for older (train+test) periods
  • adstocks (decay traces)

Revives the old R tool curves.saver.r (phrm.curves.save) that wrote output/summary/<brand>.xlsx with sheets dt.resp, dt.prof, dt.roi, dt.resp.long, dt.prof.long, dt.roi.long, setup.

 37def _derive_plan_scope_template(
 38    meta_model: MetaModel,
 39    ctx: ComputeContext,
 40    summary_node: str | None,
 41    plan_year: int | None,
 42) -> OptionScope:
 43    """
 44    Pick the target-brand plan scope used as the template for brand cloning.
 45
 46    The target brand is ``meta_model.nodes[summary_node].model.conf.target_superbrand``.
 47    Candidates are the context scopes matching that brand (``match=('superbrand',
 48    target)`` — a scope with an explicit superbrand filter). The candidate with the
 49    max ``start_year`` is chosen; when several share the max year (multi-region
 50    layout), ``plan_year`` is required. No candidate at all raises FermatricaError.
 51
 52    :param meta_model: MetaModel DAG
 53    :param ctx: ComputeContext holding the scopes
 54    :param summary_node: node providing the target brand; ``None`` resolves to the
 55        single sink node via ``_last_node_name``
 56    :param plan_year: explicit plan year to disambiguate multiple max-year candidates
 57    :return: the template OptionScope
 58    :raises FermatricaError: on missing target brand, no matching scope, or an
 59        unresolved ambiguity
 60    """
 61    summary_node = summary_node or _last_node_name(meta_model)
 62    target = getattr(meta_model.nodes[summary_node].model.conf, "target_superbrand", None)
 63    if not target:
 64        fermatrica_error(
 65            f"Node '{summary_node}' has no target_superbrand; cannot derive "
 66            "the plan scope template."
 67        )
 68
 69    candidates = filter_scopes(ctx.scopes, match=("superbrand", target))
 70    if not candidates:
 71        fermatrica_error(
 72            f"No scope in the context matches superbrand '{target}'. "
 73            "Cannot derive the plan scope template for brand summaries."
 74        )
 75
 76    if plan_year is not None:
 77        matched = [s for s in candidates if s.start_year == plan_year]
 78        if not matched:
 79            fermatrica_error(
 80                f"No target-brand scope for plan_year={plan_year} among "
 81                f"{[s.scope_name for s in candidates]}."
 82            )
 83        return matched[0]
 84
 85    max_year = max(s.start_year for s in candidates)
 86    at_max = [s for s in candidates if s.start_year == max_year]
 87    if len(at_max) > 1:
 88        fermatrica_error(
 89            f"Multiple target-brand plan scopes share the max year {max_year}: "
 90            f"{[s.scope_name for s in at_max]}. Pass plan_year to disambiguate."
 91        )
 92    return at_max[0]

 95def _derive_brand_scopes(
 96    template: OptionScope,
 97    brand: str,
 98    seasonality_mode: str | None,
 99) -> list[OptionScope]:
100    """
101    Clone the template scope for one brand.
102
103    The derived scope keeps all template attributes (dates, plan period, inflation,
104    reference period) but gets a renamed ``scope_name`` (target brand replaced in the
105    name, or ``<name>_<brand>`` appended when absent) and the ``superbrand`` filter
106    swapped to ``[brand]`` (other filter columns are kept as-is).
107
108    ``seasonality_mode='inherit'`` (default) reuses the template's seasonality
109    distribution: when the template has no explicit ``seasonality`` key,
110    ``_apply_media_ssn`` falls back to the scope name, so the derived scope points at
111    the template's scope name. ``None`` disables seasonality explicitly.
112
113    :param template: target-brand plan scope to clone
114    :param brand: brand to clone the scope for
115    :param seasonality_mode: ``'inherit'`` (default) or ``None``
116    :return: single-element list with the derived scope
117    :raises FermatricaError: on an unknown ``seasonality_mode``
118    """
119    target_vals = template.filters.get("superbrand", [])
120    target = str(target_vals[0]) if target_vals else ""
121
122    s = template.copy(deep=True)
123    if target and target in s.scope_name:
124        s.scope_name = s.scope_name.replace(target, brand)
125    else:
126        s.scope_name = f"{s.scope_name}_{brand}"
127    s.filters = {
128        col: ([brand] if col == "superbrand" else vals)
129        for col, vals in template.filters.items()
130    }
131
132    if seasonality_mode == "inherit":
133        s.seasonality = template.seasonality or template.scope_name
134    elif seasonality_mode is None:
135        s.seasonality = None
136    else:
137        fermatrica_error(
138            f"Unknown seasonality_mode '{seasonality_mode}'. Use 'inherit' or None."
139        )
140
141    return [s]

144def _non_superbrand_filters(filters: dict[str, list[str]]) -> dict[str, str]:
145    """The scope's entity filter minus the ``superbrand`` dimension.
146
147    Returns ``{column: single_value}`` (one value per non-superbrand column).
148    """
149    return {
150        col: str(vals[0])
151        for col, vals in filters.items()
152        if col != "superbrand" and vals
153    }

156def _derive_clone_brand(
157    meta_model: MetaModel,
158    ctx: ComputeContext,
159    summary_node: str,
160    brand: str,
161    plan_year: int | None,
162    seasonality_mode: str | None,
163) -> tuple[dict[str, str], str, list[OptionScope]]:
164    """Fallback clone path for brands without context plan scopes.
165
166    Returns ``(entity, template_scope_name, derived_scopes)`` where the
167    entity is the derived scope's non-superbrand filter combo (the clone
168    keeps the template's non-superbrand filter columns, so a regional
169    template yields a regional entity). Shared by ``summary_data`` and
170    ``export_summaries`` so the two call sites agree.
171    """
172    template = _derive_plan_scope_template(meta_model, ctx, summary_node, plan_year)
173    derived = _derive_brand_scopes(template, brand, seasonality_mode)
174    return _non_superbrand_filters(derived[0].filters), template.scope_name, derived

177def _brand_plan_scopes(
178    ctx: ComputeContext,
179    brand: str,
180    plan_year: int | None,
181) -> list[OptionScope]:
182    """Plan scopes of one brand: scopes matching the brand on the
183    ``superbrand`` dimension for the plan year.
184
185    ``plan_year=None`` resolves to the max start year among the matching
186    scopes (ties across regions are fine — they become distinct entities).
187    """
188    candidates = filter_scopes(ctx.scopes, match=("superbrand", brand))
189    if not candidates:
190        return []
191    year = plan_year if plan_year is not None else max(s.start_year for s in candidates)
192    return filter_scopes(candidates, start_year=year)

195def _brand_entities(brand_plan_scopes: list[OptionScope]) -> list[dict[str, str]]:
196    """Distinct non-superbrand filter combos among the brand's plan scopes.
197
198    Each entity is ``{column: value}``; the combos are sorted for
199    deterministic file naming.
200    """
201    entities = {frozenset(_non_superbrand_filters(s.filters).items()) for s in brand_plan_scopes}
202    return [dict(e) for e in sorted(entities, key=lambda e: sorted(e))]

205def _entity_plan_scopes(
206    brand_plan_scopes: list[OptionScope],
207    entity: dict[str, str],
208) -> list[OptionScope]:
209    """Plan scopes whose non-superbrand filter dict *equals* ``entity``.

211    Equality (not ``s.match``) — a national scope must not double as a

214    return [s for s in brand_plan_scopes if _non_superbrand_filters(s.filters) == entity]

239def _parent_scopes(
240    ctx: ComputeContext,
241    meta_model: MetaModel,
242    entity: dict[str, str],
243    entity_scopes: list[OptionScope],
244    plan_year: int | None,
245) -> list[OptionScope]:
246    """Parent (wider) grid scopes for a partly-regional DAG.
247
248    For every node that has ``scope_names`` but none of the entity's plan
249    scopes, the node's own context scopes (plan year) whose non-superbrand
250    filter is a **strict subset** of the entity's are added, so the node
251    still gets a grid scope (p08 ``uba``: ``brand_1_2026``, ``{}`` ⊂
252    ``{region: Cairo}``). Deduplicated, input order preserved.
253    """
254    year = plan_year if plan_year is not None else max(s.start_year for s in entity_scopes)
255    entity_names = {s.scope_name for s in entity_scopes}
256
257    parents: list[OptionScope] = []
258    seen: set[str] = set()
259    for node in meta_model.nodes.values():
260        if node.scope_names is None or entity_names & node.scope_names:
261            continue
262        for s in ctx.scopes:
263            if (
264                s.scope_name not in node.scope_names
265                or s.start_year != year
266                or s.scope_name in seen
267            ):
268                continue
269            if _filters_strict_subset(_non_superbrand_filters(s.filters), entity):
270                parents.append(s)
271                seen.add(s.scope_name)
272    return parents

275def _brand_stats(node, brand: str, filters: dict | None = None) -> pd.DataFrame | None:
276    """
277    Per-brand statistics derived from the node dataset (long variable|value frame).
278
279    Rows: ``sku_count`` (unique ``bs_key``), ``date_min``/``date_max``, ``n_rows``,
280    ``price_mean``/``price_median`` (when a price var exists), and
281    ``observed_sum``/``predicted_sum`` on ``listed.isin([2, 3])`` (when present).
282    Returns ``None`` when the brand has no rows in ``node.ds``.
283    """
284    ds = _filter_df(node.ds, brands=[brand], filters=filters)
285    if ds.empty:
286        logging.warning(f"brand_summary: brand '{brand}' has no rows in node ds — brand_stats omitted")
287        return None
288
289    stats: dict[str, object] = {}
290    if "bs_key" in ds.columns:
291        stats["sku_count"] = int(ds["bs_key"].nunique())
292    if "date" in ds.columns:
293        stats["date_min"] = pd.Timestamp(ds["date"].min())
294        stats["date_max"] = pd.Timestamp(ds["date"].max())
295    stats["n_rows"] = len(ds)
296    price_var = _get_price_var(node.model.conf)
297    if price_var and price_var in ds.columns:
298        stats["price_mean"] = ds[price_var].mean()
299        stats["price_median"] = ds[price_var].median()
300    if "listed" in ds.columns:
301        mask = ds["listed"].isin([2, 3])
302        for col in ("observed", "predicted"):
303            if col in ds.columns:
304                stats[f"{col}_sum"] = ds.loc[mask, col].sum()
305
306    return pd.DataFrame({"variable": list(stats.keys()), "value": list(stats.values())})

309def _fit_table(
310    node,
311    brand: str,
312    period: str,
313    filters: dict | None = None,
314) -> pd.DataFrame | None:
315    """
316    Fit (observed vs predicted) wide table for one brand, grouped by period.
317
318    Requires ``node.dt_pred`` (returns ``None`` with a warning otherwise). Error
319    interval columns (``*_lower`` / ``*_upper``) are dropped.
320    """
321    if node.dt_pred is None:
322        logging.warning(
323            f"brand_summary: node '{node.model.conf.Y_var}' has no dt_pred — fit sheet omitted"
324        )
325        return None
326    wide = fit_main_data(
327        node,
328        period=period,
329        show_future=False,
330        show_pretrain=False,
331        brands=[brand],
332        filters=filters,
333    )[0]
334    drop = [c for c in wide.columns if c.endswith("_lower") or c.endswith("_upper")]
335    if drop:
336        wide = wide.drop(columns=drop)
337    return wide

340def _decomposition_table(
341    node,
342    brand: str,
343    period: str,
344    listed_filter: tuple | list,
345    col: str = "value",
346    filters: dict | None = None,
347    meta_model: MetaModel | None = None,
348    node_name: str | None = None,
349    whole_graph: bool = False,
350) -> pd.DataFrame | None:
351    """
352    Dynamic decomposition for one brand, wide by variable, aggregated by period.
353
354    ``extract_effect`` output is floored to the period start, summed per
355    ``(date, variable)`` on ``col`` (``'value'`` for volume, ``'value_rub'`` for
356    value), and pivoted wide. Returns ``None`` when the brand has no effects.
357    With ``whole_graph=True`` the link factors are expanded through the DAG
358    (see ``extract_effect(whole_graph=...)``).
359    """
360    if whole_graph:
361        split = extract_effect(
362            meta_model, whole_graph=node_name, listed_filter=listed_filter,
363            brands=[brand], filters=filters,
364        )
365    else:
366        split = extract_effect(
367            node, listed_filter=listed_filter, brands=[brand], filters=filters
368        )
369    if split is None or split.empty:
370        logging.warning(
371            f"brand_summary: empty decomposition for node '{node.model.conf.Y_var}', brand '{brand}' — omitted"
372        )
373        return None
374    split = split.copy()
375    split["date"] = date_to_period(split["date"], period)
376    agg = split.groupby(["date", "variable"], as_index=False)[col].sum()
377    wide = agg.pivot(index="date", columns="variable", values=col).reset_index()
378    wide.columns.name = None
379    wide = wide.sort_values("date").reset_index(drop=True)
380    return wide

383def _curves_tables(curves_node_df: pd.DataFrame | None):
384    """
385    Incremental curves (short- and long-term) from one node's grid DataFrame.
386
387    The ``"zero"`` grid row is the baseline; per non-zero option row:
388    ``incremental_volume``/``incremental_value`` vs the baseline,
389    ``profit = incremental_value - bdg * 1e6``, ``roi = incremental_value / (bdg * 1e6)``.
390    Returns ``(short, long)`` frames with columns ``[bdg, option,
391    incremental_volume, incremental_value, profit, roi]``; ``"zero"`` rows are
392    dropped. Either frame is ``None`` when it cannot be computed (missing baseline
393    or empty grid).
394    """
395    if (
396        curves_node_df is None
397        or curves_node_df.empty
398        or "option" not in curves_node_df.columns
399        or "bdg" not in curves_node_df.columns
400    ):
401        return None, None
402
403    base_mask = curves_node_df["option"] == "zero"
404    if not base_mask.any():
405        return None, None
406
407    out = {}
408    for key, vol_col, val_col in (
409        ("short", "pred_exact_vol", "pred_exact_val"),
410        ("long", "pred_long_vol", "pred_long_val"),
411    ):
412        if vol_col not in curves_node_df.columns or val_col not in curves_node_df.columns:
413            out[key] = None
414            continue
415        base_vol = curves_node_df.loc[base_mask, vol_col]
416        base_val = curves_node_df.loc[base_mask, val_col]
417        if base_vol.empty or base_val.empty or pd.isna(base_vol.iloc[0]) or pd.isna(base_val.iloc[0]):
418            out[key] = None
419            continue
420        base_vol = float(base_vol.iloc[0])
421        base_val = float(base_val.iloc[0])
422
423        rows = []
424        for _, row in curves_node_df.loc[~base_mask].iterrows():
425            bdg = float(row["bdg"])
426            inc_vol = float(row[vol_col]) - base_vol
427            inc_val = float(row[val_col]) - base_val
428            rows.append({
429                "bdg": bdg,
430                "option": row["option"],
431                "incremental_volume": inc_vol,
432                "incremental_value": inc_val,
433                "profit": inc_val - bdg * 1e6,
434                "roi": inc_val / (bdg * 1e6) if bdg > 0 else float("nan"),
435            })
436        out[key] = pd.DataFrame(rows, columns=[
437            "bdg", "option", "incremental_volume", "incremental_value", "profit", "roi",
438        ])
439
440    return out["short"], out["long"]

443def _adstocks_table(
444    node,
445    brand: str,
446    n: int,
447    scaled: bool,
448    filters: dict | None = None,
449) -> pd.DataFrame | None:
450    """
451    Adstock decay traces for one brand, long ``step | variable | value`` frame.
452
453    Measurement columns are auto-derived from ``model.conf.params`` rows whose
454    ``fun`` contains ``adstock|dwblp``. Returns ``None`` (with a warning) when the
455    node has no adstock/dwblp params or produces no data.
456    """
457    conf = node.model.conf
458    if not hasattr(conf, "params") or conf.params is None or conf.params.empty:
459        return None
460    meas = conf.params.loc[
461        conf.params["fun"].str.contains("adstock|dwblp", na=False), "variable"
462    ].unique()
463    if len(meas) == 0:
464        logging.warning(
465            f"brand_summary: node '{conf.Y_var}' has no adstock/dwblp params — adstocks sheet omitted"
466        )
467        return None
468
469    df = adstocks_data(
470        node, brands=[brand], filters=filters, cln_meas=list(meas), n=n, if_scaled=scaled
471    )
472    if df is None or df.empty:
473        logging.warning(
474            f"brand_summary: empty adstock trace for node '{conf.Y_var}', brand '{brand}' — omitted"
475        )
476        return None
477
478    # keep only columns referenced by an active model-formula token (same rule as
479    # the adstock slide): variables present in ds but absent from the formula have
480    # no display name in the model and are excluded from the sheet
481    active_tokens = conf.model_rhs.loc[conf.model_rhs['if_active'] == 1, 'token']
482    df = df.loc[:, [active_tokens.str.contains(c).any() for c in df.columns]]
483
484    if df.shape[1] == 0:
485        logging.warning(
486            f"brand_summary: no model-formula adstock variables for node '{conf.Y_var}', brand '{brand}' — omitted"
487        )
488        return None
489
490    df = df.reset_index(drop=True)
491    df = df.reset_index(drop=False)
492    df = df.rename(columns={df.columns[0]: "step"})
493    df = df.melt(id_vars="step", var_name="variable", value_name="value")
494    return df.sort_values(["step", "variable"]).reset_index(drop=True)

525def summary_data(
526    meta_model: MetaModel,
527    ctx: ComputeContext,
528    brand: str,
529    *,
530    entity: dict[str, str] | None = None,
531    scopes: list[OptionScope] | None = None,
532    budget_step: float = 5,
533    bdg_max: int = 300,
534    cores: int | None = None,
535    if_exact: bool = True,
536    allow_past: bool = False,
537    decomp_period: str = "month",
538    decomp_listed_filter: tuple = (2, 3),
539    if_fit: bool = True,
540    adstock_n: int = 50,
541    adstock_scaled: bool = True,
542    seasonality_mode: str | None = "inherit",
543    plan_year: int | None = None,
544    summary_node: str | None = None,
545    whole_graph: bool = False,
546) -> dict[str, pd.DataFrame]:
547    """
548    Build the full sheet dict for ONE (brand, entity).
549
550    Scopes are derived from the brand's context plan scopes (``scopes=None``,
551    default) or taken from the ``scopes`` override as-is (verbatim, no entity
552    logic). The MetaModel is **deep-copied** before ``compute_grid`` (its
553    sequential branch mutates the original), so the caller's ``meta_model``
554    (used for the decomposition / fit / adstocks / setup sheets, and for
555    subsequent brands/entities) is never corrupted.
556
557    Sheets that cannot be produced (no adstock variables, no ``dt_pred``, empty
558    brand filter, empty grid) are omitted with a ``logging.warning``.
559
560    :param meta_model: MetaModel DAG (in-memory; ``node.ds`` / ``node.dt_pred``
561        reflect the reporting script's ``mm.compute(option, scopes)`` state)
562    :param ctx: ComputeContext providing scopes / seasonality / media prices
563    :param brand: brand (``superbrand`` value) to export
564    :param entity: entity (non-superbrand filter combo, ``{column: value}``) to
565        build the sheets for; ``None`` derives the single entity (clone path or
566        an error when the brand has multiple entities)
567    :param scopes: override — use these scopes as-is instead of deriving them
568        from the plan scopes (no entity splitting, no entity filters)
569    :param budget_step: curves grid step in millions
570    :param bdg_max: curves grid maximum total budget

752    """
753    Export one XLSX file per (brand, entity) into ``output_dir``.
754

756    split into one file per entity (non-superbrand filter combo) by default,

765

767    :param ctx: ComputeContext
768    :param brands: brands to export; ``None`` auto-derives from the sink node's ds

773        from the plan scopes (verbatim grid scopes, no entity splitting)
774    :param output_dir: output directory (created when missing)
775    :param budget_step: curves grid step in millions
776    :param bdg_max: curves grid maximum total budget
777    :param cores: worker processes for the grid (``None`` → ``cpu_count() - 1``)
778    :param if_exact: apply budgets only to the exact scope period
779    :param allow_past: allow budget translation over historical rows
780    :param decomp_period: decomposition / fit aggregation period
781    :param if_fit: include ``{n}_fit`` sheets
782    :param adstock_n: number of adstock decay trace steps
783    :param adstock_scaled: scale adstock traces by their maximum value
784    :param seasonality_mode: ``'inherit'`` (default) or ``None``
785    :param plan_year: plan year to pick a non-max plan year explicitly
786    :param summary_node: node providing the target brand (default: the DAG sink)
787    :param whole_graph: expand the decomposition sheets' link factors through the
788        DAG (see ``extract_effect(whole_graph=...)``). Forwarded to ``summary_data``.
789    :return: list of written file paths
790    :raises FermatricaError: when no brands can be derived (no ``superbrand`` column)
791    """
792    summary_node = summary_node or _last_node_name(meta_model)
793
794    if brands is None:
795        ds = meta_model.nodes[summary_node].ds
796        if "superbrand" not in ds.columns:
797            fermatrica_error(
798                f"MetaModel sink node '{summary_node}' dataset has no 'superbrand' "
799                "column. Pass brands explicitly."
800            )
801        brands = sorted(ds["superbrand"].dropna().unique().tolist())
802
803    entities_set = None
804    if entities is not None:
805        entities_set = {frozenset(e.items()) for e in entities}
806
807    os.makedirs(output_dir, exist_ok=True)
808    written: list[str] = []
809
810    for brand in brands:
811        try:
812            if scopes is not None:
813                brand_entities: list[dict[str, str] | None] = [None]
814            else:
815                brand_plan_scopes = _brand_plan_scopes(ctx, brand, plan_year)
816                if not brand_plan_scopes:
817                    # clone path — one derived scope; its non-superbrand filters
818                    # are the (single) entity (kept from the template, so it may
819                    # be regional)
820                    clone_entity, _, _ = _derive_clone_brand(
821                        meta_model, ctx, summary_node, brand, plan_year, seasonality_mode
822                    )
823                    if entities_set is not None and frozenset(clone_entity.items()) not in entities_set:
824                        brand_entities = []
825                    else:
826                        brand_entities = [clone_entity]
827                else:
828                    brand_entities = _brand_entities(brand_plan_scopes)
829                    if entities_set is not None:
830                        brand_entities = [
831                            e for e in brand_entities if frozenset(e.items()) in entities_set
832                        ]
833        except Exception as e:
834            logging.warning(f"brand_summary: export failed for brand '{brand}': {e}")
835            continue
836
837        for entity in brand_entities:
838            try:
839                sheets = summary_data(
840                    meta_model,
841                    ctx,
842                    brand,
843                    entity=entity,
844                    scopes=scopes,
845                    budget_step=budget_step,
846                    bdg_max=bdg_max,
847                    cores=cores,
848                    if_exact=if_exact,
849                    allow_past=allow_past,
850                    decomp_period=decomp_period,
851                    if_fit=if_fit,
852                    adstock_n=adstock_n,
853                    adstock_scaled=adstock_scaled,
854                    seasonality_mode=seasonality_mode,
855                    plan_year=plan_year,
856                    summary_node=summary_node,
857                    whole_graph=whole_graph,
858                )
859            except Exception as e:
860                logging.warning(
861                    f"brand_summary: export failed for brand '{brand}', "
862                    f"entity '{entity}': {e}"
863                )
864                continue
865
866            if not sheets:
867                logging.warning(
868                    f"brand_summary: no sheets produced for brand '{brand}', "
869                    f"entity '{entity}' — file skipped"
870                )
871                continue
872
873            label = _entity_label(entity) if entity else ""
874            stem = _sanitize_filename(brand)
875            if label:
876                stem += "_" + _sanitize_filename(label)
877            path = os.path.join(output_dir, f"{stem}.xlsx")
878            _write_workbook(path, sheets)
879            written.append(path)
880
881    return written
