 20def _compute_score(results, meta_model, budget_scope, if_long, if_volume,
 21                   error_score, summary_nodes=None, baseline_scores=None):
 22    score_key = _get_score_key(if_long, if_volume)
 23
 24    if summary_nodes is None:
 25        last_node = _last_node_name(meta_model)
 26        score_val = results[last_node][budget_scope].get(score_key, 0)
 27        if score_val is None or math.isinf(score_val):
 28            return error_score
 29        return score_val
 30
 31    combined = 0.0
 32    for node_name, (weight, node_scope) in summary_nodes.items():
 33        score_val = results[node_name][node_scope].get(score_key, 0)
 34        if score_val is None or math.isinf(score_val):
 35            return error_score
 36        combined += weight * score_val / baseline_scores[node_name]
 37    return combined

 40def _combine_trg(trg: float | dict[str, float], summary_nodes, baseline_scores):
 41    if summary_nodes is None:
 42        if isinstance(trg, dict):
 43            fermatrica_error("trg must be a float when summary_nodes is None")
 44        return float(trg)
 45
 46    if not isinstance(trg, dict):
 47        fermatrica_error(
 48            "trg must be a dict[str, float] when summary_nodes is set. "
 49            "Provide per-node targets, e.g. {'sales': 200, 'awareness': 35}."
 50        )
 51
 52    combined = 0.0
 53    for node_name, (weight, _node_scope) in summary_nodes.items():
 54        node_trg = trg.get(node_name)
 55        if node_trg is None:
 56            fermatrica_error(
 57                f"trg dict missing key '{node_name}'. "
 58                f"Required keys: {list(summary_nodes.keys())}"
 59            )
 60        if not isinstance(node_trg, (int, float)):
 61            fermatrica_error(f"trg['{node_name}'] must be a number, got {type(node_trg)}")
 62
 63        bs = baseline_scores[node_name]
 64        if bs == 0 or math.isnan(bs):
 65            fermatrica_error(
 66                f"Baseline score for node '{node_name}' is zero/NaN. Got: {bs}"
 67            )
 68        combined += weight * node_trg / bs
 69    return combined

 72class ObjectiveBudgetNG:
 73    """
 74    Wrapper for Nevergrad objective function that applies a budget
 75    constraint penalty.
 76
 77    Used with CMA-ES optimization via Nevergrad. Wraps an underlying
 78    ``objective_fun`` and adds a large penalty when the total budget
 79    deviates from ``bdg_size`` by more than 1.0 unit. This enforces the
 80    budget equality constraint through the objective itself rather than
 81    through a separate constraint mechanism.
 82
 83    Parameters
 84    ----------
 85    param_names : list[str]
 86        Ordered list of parameter names (budget item keys).
 87    objective_fun : callable
 88        Underlying objective function with signature
 89        ``f(params: list[float]) -> float``. Should return the KPI
 90        score (higher is better); the wrapper handles sign flipping.
 91    bdg_size : float
 92        Target total budget size. The sum of all parameters must equal
 93        this value.
 94    bdg_penalty : float, optional
 95        Penalty multiplier applied to budget violations.
 96        Default ``1e6``.
 97    """
 98
 99    def __init__(self
100                    , param_names
101                    , objective_fun
102                    , bdg_size
103                    , bdg_penalty=1e6):
104
105        self.param_names = param_names
106        self.objective_fun = objective_fun
107        self.bdg_size = bdg_size
108        self.bdg_penalty = bdg_penalty
109
110    def __call__(self, **params_dict):
111        params = [params_dict[par] for par in self.param_names]
112
113        # Apply budget constraint penalty
114        total = sum(params)
115        budget_violation = abs(total - self.bdg_size)
116
117        if budget_violation > 1.0:  # tolerance
118            return self.objective_fun(params) + self.bdg_penalty * budget_violation
119
120        return self.objective_fun(params)

123class ObjectiveTargetNG:
124    """
125    Wrapper for Nevergrad objective function that minimizes budget while
126    achieving a target KPI.
127
128    Used with CMA-ES optimization via Nevergrad. Minimizes the total
129    budget (sum of parameters) while penalizing deviations from the
130    target KPI. The underlying ``constraint_fun`` returns
131    ``abs(abs(score) - target)`` — the objective adds a large penalty
132    when this exceeds the tolerance.
133
134    Parameters
135    ----------
136    param_names : list[str]
137        Ordered list of parameter names (budget item keys).
138    constraint_fun : callable
139        Constraint function with signature
140        ``f(params: list[float]) -> float``. Returns the KPI violation
141        as ``abs(abs(score) - target)``.
142    trg : float
143        Combined target value. With ``summary_nodes`` this is
144        ``sum(weight * raw_target / baseline)``.
145    trg_tol_ratio : float, optional
146        Tolerance ratio relative to target. The KPI must be within
147        ``trg_tol_ratio * trg`` of the target.
148        Default ``0.0001``.
149    trg_penalty : float, optional
150        Penalty multiplier for KPI violations. Default ``1e6``.
151    """
152
153    def __init__(self
154                    , param_names
155                    , constraint_fun
156                    , trg: float
157                    , trg_tol_ratio: float = 0.0001
158                    , trg_penalty: float = 1e6):
159
160        self.param_names = param_names
161        self.constraint_fun = constraint_fun
162        self.trg = trg
163        self.trg_tol = trg_tol_ratio * trg
164        self.trg_penalty = trg_penalty
165
166    def __call__(self, **params_dict):
167        params = [params_dict[par] for par in self.param_names]
168
169        # Objective: minimize budget (sum of params)
170        budget = sum(params)
171
172        # Constraint: achieve target KPI
173        # constraint_fun returns abs(abs(score) - self.trg)
174        kpi_violation = self.constraint_fun(params)
175
176        if kpi_violation > self.trg_tol:
177            return budget + self.trg_penalty * kpi_violation
178
179        return budget

524
525        if isinstance(params_cur, pd.Series):
526            params_cur = params_cur.values
527
528        if params_cur is not None:
529            option_dict = dict_multifill(self.option_template.copy(), self.borders.columns, params_cur)
530        else:
531            option_dict = self.option_template
532
533        if self.summary_nodes is not None and self._option_budget_data is not None:
534            budget_data = dict(self._option_budget_data)

536        else:
537            budget_data = {self.budget_scope: option_dict}

543                               self.if_volume, self.error_score,
544                               self.summary_nodes, self._baseline_scores)
545
546        if score == self.error_score:
547            return self.error_score
548
549        if self.verbose:

552                print('Total budget current : ' + str(sum(params_cur)))
553                print('Scoring : ' + str(score_print) + ' ; step : ' + str(
554                    self.iter_idx) + ' : ' + datetime.datetime.now().isoformat(sep=" ", timespec="seconds"))
555            self.iter_idx += 1
556
557        return abs(abs(score) - self.trg)

Inequality constraint function callable class for target optimization with NLopt COBYLA.

Evaluates a budget allocation by running it through the MetaModel and returning abs(abs(score) - target) — the absolute distance from the target KPI. The COBYLA optimizer treats this as an inequality constraint to be kept within tolerance.
